enum ApiProtocol {
  routerOsApi8728,
  restApiHttp,
  restApiHttps,
}

extension ApiProtocolExt on ApiProtocol {
  String get label {
    switch (this) {
      case ApiProtocol.routerOsApi8728:
        return 'RouterOS API (Port 8728)';
      case ApiProtocol.restApiHttp:
        return 'REST API HTTP (Port 80)';
      case ApiProtocol.restApiHttps:
        return 'REST API HTTPS (Port 443)';
    }
  }

  int get defaultPort {
    switch (this) {
      case ApiProtocol.routerOsApi8728:
        return 8728;
      case ApiProtocol.restApiHttp:
        return 80;
      case ApiProtocol.restApiHttps:
        return 443;
    }
  }
}

enum AppRole {
  user,
  technician,
}

enum LoginRolePreference {
  auto,
  technician,
  user,
}

extension LoginRolePreferenceExt on LoginRolePreference {
  String get label {
    switch (this) {
      case LoginRolePreference.auto:
        return 'Otomatis (Sesuai Izin Akun)';
      case LoginRolePreference.technician:
        return 'Masuk sebagai Teknisi';
      case LoginRolePreference.user:
        return 'Masuk sebagai User';
    }
  }
}

class RouterProfile {
  final String id;
  final String label;
  final String host;
  final int port;
  final String username;
  final String password;
  final ApiProtocol protocol;
  final LoginRolePreference rolePreference;

  RouterProfile({
    required this.id,
    required this.label,
    required this.host,
    required this.port,
    required this.username,
    required this.password,
    required this.protocol,
    this.rolePreference = LoginRolePreference.auto,
  });

  Map<String, dynamic> toJson() => {
        'id': id,
        'label': label,
        'host': host,
        'port': port,
        'username': username,
        'password': password,
        'protocol': protocol.name,
        'rolePreference': rolePreference.name,
      };

  factory RouterProfile.fromJson(Map<String, dynamic> json) {
    return RouterProfile(
      id: json['id'] ?? DateTime.now().millisecondsSinceEpoch.toString(),
      label: json['label'] ?? json['host'] ?? 'Router',
      host: json['host'] ?? '',
      port: json['port'] ?? 8728,
      username: json['username'] ?? '',
      password: json['password'] ?? '',
      protocol: ApiProtocol.values.firstWhere(
        (e) => e.name == json['protocol'],
        orElse: () => ApiProtocol.routerOsApi8728,
      ),
      rolePreference: LoginRolePreference.values.firstWhere(
        (e) => e.name == json['rolePreference'],
        orElse: () => LoginRolePreference.auto,
      ),
    );
  }
}

class UserRoleInfo {
  final String username;
  final String groupName;
  final List<String> policies;
  final AppRole role;
  final bool canWrite;
  final bool hasFull;
  final bool canReboot;
  final bool canManageUsers;
  final bool canRunTest;

  UserRoleInfo({
    required this.username,
    required this.groupName,
    required this.policies,
    required this.role,
    required this.canWrite,
    required this.hasFull,
    required this.canReboot,
    required this.canManageUsers,
    required this.canRunTest,
  });

  bool get isTechnician => role == AppRole.technician;
  bool get isUser => role == AppRole.user;
  bool get isReadOnly => !canWrite;

  String get roleBadge {
    if (isTechnician) {
      return hasFull ? 'TEKNISI (FULL)' : 'TEKNISI (WRITE)';
    }
    return 'MODE USER';
  }
}

  final IconData icon;
  final List<WinboxSubMenu> subMenus;

class WinboxSubMenu {
  final String title;
  final String path;
  final String primaryField;
  final String? secondaryField;
  final List<String> displayColumns;
  const WinboxSubMenu(this.title, this.path, this.primaryField,
      this.secondaryField,
      [this.displayColumns = const []]);
}

class WinboxMenuItemReference {
  final int categoryIndex;
  final int subMenuIndex;
  final WinboxCategory category;
  final WinboxSubMenu subMenu;
  const WinboxMenuItemReference({
    required this.categoryIndex,
    required this.subMenuIndex,
    required this.category,
    required this.subMenu,
  });
}

WinboxMenuItemReference? findWinboxMenuByPath(String path) {
  for (int c = 0; c < winboxCategories.length; c++) {
    for (int s = 0; s < winboxCategories[c].subMenus.length; s++) {
      if (winboxCategories[c].subMenus[s].path == path) {
        return WinboxMenuItemReference(
          categoryIndex: c,
          subMenuIndex: s,
          category: winboxCategories[c],
          subMenu: winboxCategories[c].subMenus[s],
        );
      }
    }
  }
  return null;
}

List<WinboxMenuItemReference> searchWinboxMenus(String query) {
  final q = query.trim().toLowerCase();
  final results = <WinboxMenuItemReference>[];
  for (int c = 0; c < winboxCategories.length; c++) {
    final cat = winboxCategories[c];
    for (int s = 0; s < cat.subMenus.length; s++) {
      final sub = cat.subMenus[s];
      if (q.isEmpty ||
          cat.title.toLowerCase().contains(q) ||
          sub.title.toLowerCase().contains(q) ||
          sub.path.toLowerCase().contains(q)) {
        results.add(WinboxMenuItemReference(
          categoryIndex: c,
          subMenuIndex: s,
          category: cat,
          subMenu: sub,
        ));
      }
    }
  }
  return results;
}

import 'dart:async';
import 'dart:io';
import 'package:network_info_plus/network_info_plus.dart';
import 'package:permission_handler/permission_handler.dart';

/// Hasil deteksi router MikroTik pada jaringan lokal
class MikrotikDiscoveryResult {
  /// Alamat IP Gateway / Router MikroTik (misal: 192.168.88.1 atau 192.168.1.1)
  final String gatewayIp;

  /// Port API terdeteksi yang siap digunakan untuk koneksi (prioritas: 8728 -> 8777 -> 80)
  final int apiPort;

  /// Daftar port signature MikroTik yang terbuka
  final List<int> openPorts;

  /// Indikator apakah port Winbox (8291) terbuka
  final bool hasWinbox;

  /// Nama SSID Wi-Fi aktif (jika diizinkan sistem)
  final String? wifiSsid;

  const MikrotikDiscoveryResult({
    required this.gatewayIp,
    required this.apiPort,
    required this.openPorts,
    required this.hasWinbox,
    this.wifiSsid,
  });

  @override
  String toString() =>
      'MikrotikDiscoveryResult(gateway: $gatewayIp, apiPort: $apiPort, openPorts: $openPorts, winbox: $hasWinbox, ssid: $wifiSsid)';
}

/// Zero-Touch Auto-Discovery Engine untuk MikroTik Router
class MikrotikAutoDiscovery {
  final NetworkInfo _networkInfo;

  MikrotikAutoDiscovery({NetworkInfo? networkInfo})
      : _networkInfo = networkInfo ?? NetworkInfo();

  /// Port signature MikroTik untuk validasi keberadaan router:
  /// - 8291: Winbox / MAC-Telnet (Indikator terkuat perangkat MikroTik)
  /// - 8728: RouterOS API biner standar
  /// - 8777: Port API alternatif umum yang sering disetel oleh ISP
  /// - 80: WebFig HTTP (Web interface RouterOS)
  static const int portWinbox = 8291;
  static const int portApiDefault = 8728;
  static const int portApiCustom = 8777;
  static const int portWebFig = 80;

  static const List<int> probePorts = [
    portWinbox,
    portApiDefault,
    portApiCustom,
    portWebFig,
  ];

  /// Menjalankan pemindaian Zero-Touch di latar belakang.
  /// Mengembalikan [MikrotikDiscoveryResult] jika router MikroTik terdeteksi,
  /// atau `null` jika tidak terhubung Wi-Fi, port tertutup, atau terjadi kegagalan.
  Future<MikrotikDiscoveryResult?> discoverGatewayRouter({
    Duration timeoutPerPort = const Duration(seconds: 2),
  }) async {
    try {
      // 1. Dapatkan IP Gateway Wi-Fi
      final gatewayIp = await _resolveGatewayIp();
      if (gatewayIp == null || gatewayIp.isEmpty || gatewayIp == '0.0.0.0') {
        return null;
      }

      // Ambil SSID Wi-Fi opsional
      String? ssid;
      try {
        ssid = await _networkInfo.getWifiName();
        if (ssid != null) {
          // Bersihkan tanda petik ganda yang sering disertakan oleh iOS/Android
          ssid = ssid.replaceAll('"', '').trim();
          if (ssid.isEmpty) ssid = null;
        }
      } catch (_) {}

      // 2. Lakukan probe TCP port signature secara paralel dan non-blocking
      final portFutures = probePorts.map((port) => _probePort(gatewayIp, port, timeoutPerPort));
      final probeResults = await Future.wait(portFutures);

      final openPorts = <int>[];
      for (final res in probeResults) {
        if (res.isOpen) {
          openPorts.add(res.port);
        }
      }

      final hasWinbox = openPorts.contains(portWinbox);
      final hasApiDefault = openPorts.contains(portApiDefault);
      final hasApiCustom = openPorts.contains(portApiCustom);

      // 3. Evaluasi Signature MikroTik:
      // Harus memiliki Port 8291 (Winbox) ATAU salah satu port API (8728 / 8777)
      if (hasWinbox || hasApiDefault || hasApiCustom) {
        // Tentukan port API terbaik untuk form login
        final chosenApiPort = hasApiDefault
            ? portApiDefault
            : (hasApiCustom ? portApiCustom : (openPorts.contains(portWebFig) ? portWebFig : portApiDefault));

        return MikrotikDiscoveryResult(
          gatewayIp: gatewayIp,
          apiPort: chosenApiPort,
          openPorts: openPorts,
          hasWinbox: hasWinbox,
          wifiSsid: ssid,
        );
      }

      return null;
    } catch (_) {
      // Silent fail tanpa mengganggu pengguna
      return null;
    }
  }

  /// Meminta izin lokasi di Android (diperlukan untuk membaca info Wi-Fi pada Android 10+)
  Future<bool> _requestLocationPermissionIfNeeded() async {
    try {
      if (Platform.isAndroid) {
        var status = await Permission.locationWhenInUse.status;
        if (!status.isGranted) {
          status = await Permission.locationWhenInUse.request();
        }
        return status.isGranted;
      }
    } catch (_) {}
    return true;
  }

  /// Menyelesaikan IP Gateway melalui network_info_plus dengan fallback ke NetworkInterface lokal
  Future<String?> _resolveGatewayIp() async {
    // Coba minta izin lokasi terlebih dahulu jika di Android
    await _requestLocationPermissionIfNeeded();

    String? gateway;
    try {
      gateway = await _networkInfo.getWifiGatewayIP();
      if (gateway != null && gateway.trim().isNotEmpty && gateway != '0.0.0.0') {
        return gateway.trim();
      }
    } catch (_) {}

    // Fallback: Analisis alamat IP interface lokal jika network_info_plus gagal mendapatkan gateway
    try {
      final interfaces = await NetworkInterface.list(
        type: InternetAddressType.IPv4,
        includeLinkLocal: false,
      );

      for (final iface in interfaces) {
        final nameLower = iface.name.toLowerCase();
        // Hanya cek interface yang berasosiasi dengan Wi-Fi / WLAN
        if (nameLower.contains('wlan') || nameLower.contains('wi-fi') || nameLower.contains('en0')) {
          for (final addr in iface.addresses) {
            final ip = addr.address;
            if (ip == '127.0.0.1' || ip.startsWith('169.254.')) continue;
            final parts = ip.split('.');
            if (parts.length == 4) {
              return '${parts[0]}.${parts[1]}.${parts[2]}.1';
            }
          }
        }
      }
    } catch (_) {}

    return null;
  }

  /// Probe satu port TCP dengan batas timeout ketat (2 detik).
  /// Menangani SocketException dan TimeoutException secara aman.
  Future<_PortProbeResult> _probePort(String host, int port, Duration timeout) async {
    Socket? socket;
    try {
      socket = await Socket.connect(host, port, timeout: timeout);
      socket.destroy();
      return _PortProbeResult(port: port, isOpen: true);
    } catch (_) {
      return _PortProbeResult(port: port, isOpen: false);
    } finally {
      try {
        socket?.destroy();
      } catch (_) {}
    }
  }
}

class _PortProbeResult {
  final int port;
  final bool isOpen;
  const _PortProbeResult({required this.port, required this.isOpen});
}

import 'dart:async';
import 'dart:convert';
import 'dart:io';
import '../models/router_models.dart';

class MikroTikService {
  final RouterProfile profile;
  _RouterOsBinaryClient? _persistentBinaryClient;
  Future<void> _commandQueue = Future.value();
  bool _isDisposed = false;

  final _connectionStateController = StreamController<bool>.broadcast();
  Stream<bool> get connectionStateStream => _connectionStateController.stream;
  bool _lastKnownConnected = true;
  bool get isConnected => _lastKnownConnected;

  void _notifyConnectionState(bool connected) {
    if (_lastKnownConnected != connected) {
      _lastKnownConnected = connected;
      if (!_connectionStateController.isClosed) {
        _connectionStateController.add(connected);
      }
    }
  }

  MikroTikService(this.profile);

  /// Menutup koneksi persistent socket saat keluar atau ganti router
  Future<void> dispose() async {
    _isDisposed = true;
    await _persistentBinaryClient?.close();
    _persistentBinaryClient = null;
    if (!_connectionStateController.isClosed) {
      await _connectionStateController.close();
    }
  }

  /// Mengecek hak akses user dari /user dan /user/group di MikroTik secara serentak
  Future<UserRoleInfo> detectUserRole() async {
    final results = await Future.wait([
      _safeRunCommand('/user/print'),
      _safeRunCommand('/user/group/print'),
    ]);
    final users = results[0];
    final groups = results[1];

    final currentUser = users.firstWhere(
      (u) => u['name'] == profile.username,
      orElse: () => {'name': profile.username, 'group': 'write'},
    );

    final groupName = currentUser['group'] ?? 'read';
    final groupData = groups.firstWhere(
      (g) => g['name'] == groupName,
      orElse: () {
        final gLow = groupName.toLowerCase();
        if (gLow == 'full') return {'name': groupName, 'policy': 'full,write,read,reboot,policy,test'};
        if (gLow == 'write') return {'name': groupName, 'policy': 'write,read,reboot,test'};
        return {'name': groupName, 'policy': 'read'};
      },
    );

    final policyRaw = groupData['policy'] ?? '';
    final policies = policyRaw
        .split(',')
        .map((e) => e.trim().toLowerCase())
        .where((e) => e.isNotEmpty)
        .toList();

    if (policies.isEmpty) {
      final gLow = groupName.toLowerCase();
      if (gLow == 'full') {
        policies.addAll(['full', 'write', 'read', 'reboot', 'policy', 'test']);
      } else if (gLow == 'write') {
        policies.addAll(['write', 'read', 'reboot', 'test']);
      } else {
        policies.add('read');
      }
    }

    final hasFull = policies.contains('full');
    final canWrite = policies.contains('write') || hasFull;
    final canReboot = policies.contains('reboot') || hasFull;
    final canManageUsers = policies.contains('policy') || hasFull;
    final canRunTest = policies.contains('test') || hasFull;

    // Menentukan AppRole (User vs Teknisi) sesuai spesifikasi 2 Role
    final gNameLow = groupName.toLowerCase();
    AppRole role;

    if (profile.rolePreference == LoginRolePreference.user) {
      role = AppRole.user;
    } else if (gNameLow.contains('user') || gNameLow.contains('pelanggan') || gNameLow.contains('customer')) {
      role = AppRole.user;
    } else if (!canWrite) {
      // Jika akun router tidak memiliki izin write/full, amankan ke mode User (Read-Only)
      role = AppRole.user;
    } else {
      // Memiliki izin write/full dan tidak memilih mode user
      role = AppRole.technician;
    }

    return UserRoleInfo(
      username: profile.username,
      groupName: groupName,
      policies: policies,
      role: role,
      canWrite: canWrite,
      hasFull: hasFull,
      canReboot: canReboot,
      canManageUsers: canManageUsers,
      canRunTest: canRunTest,
    );
  }

  /// Membaca interface Wi-Fi (mendukung RouterOS v7 /interface/wifi & v6 /interface/wireless)
  Future<List<WifiInterfaceInfo>> fetchWifiInterfaces() async {
    final result = <WifiInterfaceInfo>[];

    // 1. Cek RouterOS v7 (/interface/wifi)
    final v7Wifis = await _safeRunCommand('/interface/wifi/print');
    if (v7Wifis.isNotEmpty) {
      for (final w in v7Wifis) {
        result.add(WifiInterfaceInfo(
          id: w['.id'] ?? '',
          name: w['name'] ?? 'wifi1',
          ssid: w['configuration.ssid'] ?? w['ssid'] ?? '(Belum diset)',
          password: w['security.passphrase'] ?? '',
          securityProfile: w['security'] ?? 'default',
          isRosV7Wifi: true,
          disabled: w['disabled'] == 'true',
        ));
      }
      return result;
    }

    // 2. Cek RouterOS v6 (/interface/wireless)
    final v6Wifis = await _safeRunCommand('/interface/wireless/print');
    final secProfiles = await _safeRunCommand('/interface/wireless/security-profiles/print');

    for (final w in v6Wifis) {
      final secName = w['security-profile'] ?? 'default';
      final matchedSec = secProfiles.firstWhere(
        (s) => s['name'] == secName,
        orElse: () => <String, String>{},
      );

      result.add(WifiInterfaceInfo(
        id: w['.id'] ?? '',
        name: w['name'] ?? 'wlan1',
        ssid: w['ssid'] ?? 'MikroTik',
        password: matchedSec['wpa2-pre-shared-key'] ?? matchedSec['wpa-pre-shared-key'] ?? '',
        securityProfile: secName,
        isRosV7Wifi: false,
        disabled: w['disabled'] == 'true',
      ));
    }

    return result;
  }

  /// Mengubah SSID dan Password Wi-Fi
  Future<void> updateWifiConfig({
    required WifiInterfaceInfo wifi,
    required String newSsid,
    required String newPassword,
  }) async {
    if (newPassword.isNotEmpty && newPassword.length < 8) {
      throw Exception('Password Wi-Fi minimal 8 karakter.');
    }

    if (wifi.isRosV7Wifi) {
      final params = <String, String>{
        '.id': wifi.id,
        'configuration.ssid': newSsid,
      };
      if (newPassword.isNotEmpty) {
        params['security.passphrase'] = newPassword;
        params['security.authentication-types'] = 'wpa2-psk,wpa3-psk';
      }
      await runCommand('/interface/wifi/set', params: params);
    } else {
      await runCommand('/interface/wireless/set', params: {
        '.id': wifi.id,
        'ssid': newSsid,
      });

      if (newPassword.isNotEmpty) {
        final secProfiles = await runCommand('/interface/wireless/security-profiles/print');
        final targetSec = secProfiles.firstWhere(
          (s) => s['name'] == wifi.securityProfile,
          orElse: () => <String, String>{},
        );
        if (targetSec['.id'] != null) {
          await runCommand('/interface/wireless/security-profiles/set', params: {
            '.id': targetSec['.id']!,
            'mode': 'dynamic-keys',
            'authentication-types': 'wpa2-psk',
            'wpa2-pre-shared-key': newPassword,
          });
        }
      }
    }
  }

  /// Mengambil daftar perangkat aktif (DHCP Lease + ARP + Wireless Registration) beserta status blokir
  Future<List<ConnectedDevice>> fetchConnectedDevices() async {
    final results = await Future.wait([
      _safeRunCommand('/ip/dhcp-server/lease/print'),
      _safeRunCommand('/ip/arp/print'),
      _safeRunCommand('/ip/firewall/filter/print'),
      _safeRunCommand('/interface/wireless/access-list/print'),
      _safeRunCommand('/interface/wifi/access-list/print'),
    ]);

    final leases = results[0];
    final arps = results[1];
    final firewallRules = results[2];
    final v6AccessList = results[3];
    final v7AccessList = results[4];

    final blockedMacs = <String>{};
    for (final rule in firewallRules) {
      final mac = (rule['src-mac-address'] ?? '').toUpperCase();
      if (mac.isNotEmpty && rule['action'] == 'drop' && rule['disabled'] != 'true') {
        blockedMacs.add(mac);
      }
    }
    for (final acl in v6AccessList) {
      final mac = (acl['mac-address'] ?? '').toUpperCase();
      if (mac.isNotEmpty && acl['authentication'] == 'no') {
        blockedMacs.add(mac);
      }
    }
    for (final acl in v7AccessList) {
      final mac = (acl['mac-address'] ?? '').toUpperCase();
      if (mac.isNotEmpty && acl['action'] == 'reject') {
        blockedMacs.add(mac);
      }
    }

    final deviceMap = <String, ConnectedDevice>{};

    for (final lease in leases) {
      final mac = (lease['mac-address'] ?? '').toUpperCase();
      if (mac.isEmpty) continue;
      deviceMap[mac] = ConnectedDevice(
        macAddress: mac,
        ipAddress: lease['address'] ?? lease['active-address'] ?? '-',
        hostName: (lease['host-name'] ?? '').isNotEmpty ? lease['host-name']! : 'Perangkat DHCP',
        source: 'DHCP Lease',
        isBlocked: blockedMacs.contains(mac),
      );
    }

    for (final arp in arps) {
      final mac = (arp['mac-address'] ?? '').toUpperCase();
      if (mac.isEmpty || deviceMap.containsKey(mac)) continue;
      deviceMap[mac] = ConnectedDevice(
        macAddress: mac,
        ipAddress: arp['address'] ?? '-',
        hostName: 'Host (${arp['interface'] ?? 'LAN'})',
        source: 'ARP Table',
        isBlocked: blockedMacs.contains(mac),
      );
    }

    // Pastikan perangkat yang diblokir tetap muncul meski sudah terputus dari ARP/DHCP
    for (final mac in blockedMacs) {
      if (!deviceMap.containsKey(mac)) {
        deviceMap[mac] = ConnectedDevice(
          macAddress: mac,
          ipAddress: 'Terblokir',
          hostName: 'Perangkat Diblokir',
          source: 'Firewall / ACL',
          isBlocked: true,
        );
      }
    }

    return deviceMap.values.toList();
  }

  /// Engine Static IP ISP: Menggabungkan data dari /ip/arp, /queue/simple, dan /ip/firewall/filter
  Future<List<StaticHostDevice>> fetchStaticDevices() async {
    final results = await Future.wait([
      _safeRunCommand('/ip/arp/print'),
      _safeRunCommand('/queue/simple/print'),
      _safeRunCommand('/ip/firewall/filter/print'),
    ]);

    final arpList = results[0];
    final queueList = results[1];
    final filterRules = results[2];

    // Filter drop rule yang diblokir oleh aplikasi ISP
    final blockedIps = <String>{};
    final blockedMacs = <String>{};
    for (final r in filterRules) {
      final action = r['action'] ?? '';
      final comment = r['comment'] ?? '';
      if (action == 'drop' && comment.startsWith('BLOCKED_BY_ISP_APP')) {
        final srcIp = (r['src-address'] ?? '').split('/')[0].trim();
        if (srcIp.isNotEmpty) blockedIps.add(srcIp);

        final srcMac = (r['src-mac-address'] ?? '').trim().toUpperCase();
        if (srcMac.isNotEmpty) blockedMacs.add(srcMac);
      }
    }

    // Index simple queue berdasarkan target IP (menghilangkan prefix /32)
    final queueByIp = <String, Map<String, String>>{};
    for (final q in queueList) {
      final target = q['target'] ?? '';
      final cleanTarget = target.split('/')[0].trim();
      if (cleanTarget.isNotEmpty) {
        queueByIp[cleanTarget] = q;
      }
    }

    final result = <StaticHostDevice>[];
    final seenIps = <String>{};
    final seenMacs = <String>{};

    // 1. Host aktif dari ARP table
    for (final arp in arpList) {
      final ip = (arp['address'] ?? '').trim();
      if (ip.isEmpty) continue;
      seenIps.add(ip);

      final mac = (arp['mac-address'] ?? '').trim().toUpperCase();
      if (mac.isNotEmpty) seenMacs.add(mac);

      final iface = arp['interface'] ?? 'ether';
      final arpComment = arp['comment'] ?? '';

      final q = queueByIp[ip];
      String customerName = '';
      String maxLimit = '-';
      String currentRate = '-';

      if (q != null) {
        final qComment = q['comment'] ?? '';
        final qName = q['name'] ?? '';
        if (qComment.isNotEmpty) {
          customerName = qComment;
        } else if (qName.isNotEmpty) {
          customerName = qName;
        }

        maxLimit = q['max-limit'] ?? '-';
        currentRate = q['rate'] ?? '-';
      }

      if (customerName.isEmpty && arpComment.isNotEmpty) {
        customerName = arpComment;
      }
      if (customerName.isEmpty) {
        customerName = 'Host $iface';
      }

      final isBlockedByIp = blockedIps.contains(ip);
      final isBlockedByMac = mac.isNotEmpty && blockedMacs.contains(mac);
      final isUnqueuedAnomaly = (q == null);

      result.add(StaticHostDevice(
        ipAddress: ip,
        macAddress: mac,
        interfaceName: iface,
        customerName: customerName,
        maxLimit: _formatRate(maxLimit),
        currentRate: _formatRate(currentRate),
        isBlockedByIp: isBlockedByIp,
        isBlockedByMac: isBlockedByMac,
        isUnqueuedAnomaly: isUnqueuedAnomaly,
        comment: arpComment.isNotEmpty ? arpComment : (q?['comment']),
      ));
    }

    // 2. IP yang terblokir namun sudah offline / hilang dari ARP
    for (final bIp in blockedIps) {
      if (!seenIps.contains(bIp)) {
        final q = queueByIp[bIp];
        final name = q?['comment']?.isNotEmpty == true
            ? q!['comment']!
            : (q?['name']?.isNotEmpty == true ? q!['name']! : 'Host Terblokir IP');
        result.add(StaticHostDevice(
          ipAddress: bIp,
          macAddress: '',
          interfaceName: 'Drop Rules',
          customerName: '$name (Offline)',
          maxLimit: _formatRate(q?['max-limit'] ?? '-'),
          currentRate: '-',
          isBlockedByIp: true,
          isBlockedByMac: false,
          isUnqueuedAnomaly: q == null,
          comment: 'Terblokir IP (Tidak ada di ARP)',
        ));
      }
    }

    // 3. MAC yang terblokir namun sudah offline / hilang dari ARP
    for (final bMac in blockedMacs) {
      if (!seenMacs.contains(bMac)) {
        result.add(StaticHostDevice(
          ipAddress: '-',
          macAddress: bMac,
          interfaceName: 'Drop Rules',
          customerName: 'Host Terblokir MAC (Offline)',
          maxLimit: '-',
          currentRate: '-',
          isBlockedByIp: false,
          isBlockedByMac: true,
          isUnqueuedAnomaly: false,
          comment: 'Terblokir MAC (Tidak ada di ARP)',
        ));
      }
    }

    return result;
  }

  static String _formatRate(String raw) {
    if (raw == '-' || raw.isEmpty) return '-';
    final parts = raw.split('/');
    if (parts.length == 2) {
      return '${_formatSingleRate(parts[0])} / ${_formatSingleRate(parts[1])}';
    }
    return _formatSingleRate(raw);
  }

  static String _formatSingleRate(String str) {
    final trimmed = str.trim();
    if (trimmed.isEmpty || trimmed == '0') return '0 bps';
    final val = int.tryParse(trimmed);
    if (val == null) return trimmed;
    if (val >= 1000000000) return '${(val / 1000000000).toStringAsFixed(1)} Gbps';
    if (val >= 1000000) return '${(val / 1000000).toStringAsFixed(1)} Mbps';
    if (val >= 1000) return '${(val / 1000).toStringAsFixed(0)} Kbps';
    return '$val bps';
  }

  /// Memblokir atau membuka blokir IP Address pelanggan
  Future<void> toggleBlockIp(String ipAddress, {required bool block, String? label}) async {
    final cleanIp = ipAddress.trim();
    if (cleanIp.isEmpty || cleanIp == '-') return;

    if (block) {
      final rules = await _safeRunCommand('/ip/firewall/filter/print');
      final alreadyBlocked = rules.any((r) {
        final src = (r['src-address'] ?? '').split('/')[0].trim();
        return src == cleanIp && r['action'] == 'drop';
      });

      if (!alreadyBlocked) {
        final commentText = 'BLOCKED_BY_ISP_APP${label != null && label.trim().isNotEmpty ? ' (${label.trim()})' : ''}';
        await runCommand('/ip/firewall/filter/add', params: {
          'chain': 'forward',
          'src-address': cleanIp,
          'action': 'drop',
          'comment': commentText,
        });
      }
    } else {
      final rules = await _safeRunCommand('/ip/firewall/filter/print');
      for (final r in rules) {
        final src = (r['src-address'] ?? '').split('/')[0].trim();
        if (src == cleanIp && r['action'] == 'drop' && r['.id'] != null) {
          await runCommand('/ip/firewall/filter/remove', params: {'.id': r['.id']!});
        }
      }
    }
  }

  /// Memblokir atau membuka blokir MAC Address perangkat
  Future<void> toggleBlockMac(String macAddress, {required bool block, String? label}) async {
    final cleanMac = macAddress.trim().toUpperCase();
    if (cleanMac.isEmpty || cleanMac == '-') return;

    if (block) {
      final rules = await _safeRunCommand('/ip/firewall/filter/print');
      final alreadyBlocked = rules.any((r) {
        final src = (r['src-mac-address'] ?? '').trim().toUpperCase();
        return src == cleanMac && r['action'] == 'drop';
      });

      if (!alreadyBlocked) {
        final commentText = 'BLOCKED_BY_ISP_APP${label != null && label.trim().isNotEmpty ? ' (${label.trim()})' : ''}';
        await runCommand('/ip/firewall/filter/add', params: {
          'chain': 'forward',
          'src-mac-address': cleanMac,
          'action': 'drop',
          'comment': commentText,
        });
      }
    } else {
      // Hapus rule firewall yang memblokir MAC tersebut
      final rules = await _safeRunCommand('/ip/firewall/filter/print');
      for (final r in rules) {
        if ((r['src-mac-address'] ?? '').toUpperCase() == cleanMac && r['.id'] != null) {
          await runCommand('/ip/firewall/filter/remove', params: {'.id': r['.id']!});
        }
      }
      // Hapus juga jika ada di wireless access-list
      final v6Acl = await _safeRunCommand('/interface/wireless/access-list/print');
      for (final a in v6Acl) {
        if ((a['mac-address'] ?? '').toUpperCase() == cleanMac && a['.id'] != null) {
          await runCommand('/interface/wireless/access-list/remove', params: {'.id': a['.id']!});
        }
      }
    }
  }

    } catch (_) {

    }
  }

  /// Eksekutor antrean perintah FIFO persisten untuk Binary API (8728) dan REST API (80/443)
  Future<List<Map<String, String>>> runCommand(
    String path, {
    Map<String, String>? params,
  }) {
    if (_isDisposed) {
      return Future.error(Exception('MikroTikService sudah di-dispose.'));
    }

    final completer = Completer<List<Map<String, String>>>();
    _commandQueue = _commandQueue.then((_) async {
      try {
        final result = await _executeCommand(path, params: params);
        if (!completer.isCompleted) completer.complete(result);
      } catch (e, st) {
        if (!completer.isCompleted) completer.completeError(e, st);
      }
    });
    return completer.future;
  }

  Future<List<Map<String, String>>> _executeCommand(
    String path, {
    Map<String, String>? params,
  }) async {
    if (profile.protocol == ApiProtocol.routerOsApi8728) {
      if (_persistentBinaryClient == null || _persistentBinaryClient!.isClosed) {
        final client = _RouterOsBinaryClient(
          profile.host,
          profile.port,
          onConnectionChanged: _notifyConnectionState,
        );
        await client.connectAndLogin(profile.username, profile.password);
        _persistentBinaryClient = client;
        _notifyConnectionState(true);
      }

      try {
        final words = <String>[path];
        if (params != null) {
          params.forEach((k, v) => words.add('=$k=$v'));
        }
        final res = await _persistentBinaryClient!.sendCommand(words);
        _notifyConnectionState(true);
        return res;
      } catch (e) {
        // Socket terputus atau timeout, tutup client agar perintah berikutnya melakukan auto-reconnect
        _notifyConnectionState(false);
        await _persistentBinaryClient?.close();
        _persistentBinaryClient = null;
        rethrow;
      }
    } else {
      final scheme = profile.protocol == ApiProtocol.restApiHttps ? 'https' : 'http';
      final cleanPath = path.startsWith('/') ? path : '/$path';
      final url = '$scheme://${profile.host}:${profile.port}/rest$cleanPath';

      final httpClient = HttpClient()
        ..connectionTimeout = const Duration(milliseconds: 1500)
        ..badCertificateCallback = (cert, host, port) => true;

      try {
        final isPrint = cleanPath.endsWith('/print');
        final method = isPrint ? 'GET' : 'POST';
        final targetUrl = isPrint ? url.replaceFirst(RegExp(r'/print$'), '') : url;

        final request = await httpClient.openUrl(method, Uri.parse(targetUrl));
        final auth = base64Encode(utf8.encode('${profile.username}:${profile.password}'));
        request.headers.set(HttpHeaders.authorizationHeader, 'Basic $auth');
        request.headers.set(HttpHeaders.contentTypeHeader, 'application/json');

        if (!isPrint && params != null && params.isNotEmpty) {
          request.write(jsonEncode(params));
        }

        final response = await request.close();
        final body = await response.transform(utf8.decoder).join();

        if (response.statusCode == 401) {
          _notifyConnectionState(false);
          throw Exception('Username atau password salah (401).');
        }
        if (response.statusCode >= 400) {
          throw Exception('HTTP ${response.statusCode}: $body');
        }
        _notifyConnectionState(true);
        if (body.isEmpty) return [];

        final decoded = jsonDecode(body);
        if (decoded is List) {
          return decoded
              .map((item) => Map<String, String>.from(
                    (item as Map).map((k, v) => MapEntry(k.toString(), v.toString())),
                  ))
              .toList();
        } else if (decoded is Map) {
          return [
            decoded.map((k, v) => MapEntry(k.toString(), v.toString())),
          ];
        }
        return [];
      } catch (e) {
        _notifyConnectionState(false);
        rethrow;
      } finally {
        httpClient.close();
      }
    }
  }
}

/// Client socket biner MikroTik RouterOS API (Port 8728)
class _RouterOsBinaryClient {
  final String host;
  final int port;
  final void Function(bool isConnected)? onConnectionChanged;
  Socket? _socket;
  final List<int> _buffer = [];
  final List<String> _currentSentence = [];
  final StreamController<List<String>> _controller = StreamController.broadcast();
  bool _isClosed = false;

  bool get isClosed => _isClosed;

  _RouterOsBinaryClient(this.host, this.port, {this.onConnectionChanged});

  Future<void> connectAndLogin(String username, String password) async {
    _isClosed = false;
    _socket = await Socket.connect(host, port, timeout: const Duration(milliseconds: 1500));
    _socket!.setOption(SocketOption.tcpNoDelay, true);
    onConnectionChanged?.call(true);

    _socket!.listen(
      _onData,
      onError: (e) {
        _isClosed = true;
        onConnectionChanged?.call(false);
        if (!_controller.isClosed) _controller.addError(e);
      },
      onDone: () {
        _isClosed = true;
        onConnectionChanged?.call(false);
        if (!_controller.isClosed) _controller.close();
      },
      cancelOnError: false,
    );

    await sendCommand(['/login', '=name=$username', '=password=$password']);
  }

  void _onData(List<int> data) {
    _buffer.addAll(data);
    while (true) {
      final word = _decodeNextWord();
      if (word == null) break;
      if (word.isEmpty) {
        _controller.add(List<String>.from(_currentSentence));
        _currentSentence.clear();
      } else {
        _currentSentence.add(word);
      }
    }
  }

  String? _decodeNextWord() {
    if (_buffer.isEmpty) return null;
    final b0 = _buffer[0];
    int len = 0;
    int offset = 0;

    if ((b0 & 0x80) == 0x00) {
      len = b0;
      offset = 1;
    } else if ((b0 & 0xC0) == 0x80) {
      if (_buffer.length < 2) return null;
      len = ((b0 & 0x3F) << 8) | _buffer[1];
      offset = 2;
    } else if ((b0 & 0xE0) == 0xC0) {
      if (_buffer.length < 3) return null;
      len = ((b0 & 0x1F) << 16) | (_buffer[1] << 8) | _buffer[2];
      offset = 3;
    } else if ((b0 & 0xF0) == 0xE0) {
      if (_buffer.length < 4) return null;
      len = ((b0 & 0x0F) << 24) | (_buffer[1] << 16) | (_buffer[2] << 8) | _buffer[3];
      offset = 4;
    } else {
      if (_buffer.length < 5) return null;
      len = (_buffer[1] << 24) | (_buffer[2] << 16) | (_buffer[3] << 8) | _buffer[4];
      offset = 5;
    }

    if (_buffer.length < offset + len) return null;
    final bytes = _buffer.sublist(offset, offset + len);
    _buffer.removeRange(0, offset + len);
    return utf8.decode(bytes, allowMalformed: true);
  }

  Future<List<Map<String, String>>> sendCommand(List<String> words) async {
    if (_socket == null) throw Exception('Socket belum terhubung.');
    final completer = Completer<List<Map<String, String>>>();
    final records = <Map<String, String>>[];
    String? trapError;

    late StreamSubscription<List<String>> sub;
    sub = _controller.stream.listen(
      (sentence) {
        if (sentence.isEmpty) return;
        final tag = sentence.first;
        if (tag == '!re') {
          final row = <String, String>{};
          for (final w in sentence.skip(1)) {
            if (w.startsWith('=')) {
              final idx = w.indexOf('=', 1);
              if (idx != -1) row[w.substring(1, idx)] = w.substring(idx + 1);
            }
          }
          records.add(row);
        } else if (tag == '!trap') {
          for (final w in sentence.skip(1)) {
            if (w.startsWith('=message=')) {
              trapError = w.substring('=message='.length);
            }
          }
        } else if (tag == '!done') {
          sub.cancel();
          if (trapError != null) {
            completer.completeError(Exception(trapError));
          } else {
            completer.complete(records);
          }
        } else if (tag == '!fatal') {
          sub.cancel();
          completer.completeError(Exception('Koneksi diputus router (!fatal).'));
        }
      },
      onError: (e) {
        sub.cancel();
        if (!completer.isCompleted) completer.completeError(e);
      },
    );

    for (final w in words) {
      final bytes = utf8.encode(w);
      _socket!.add(_encodeLength(bytes.length));
      _socket!.add(bytes);
    }
    _socket!.add([0x00]);
    await _socket!.flush();

    return completer.future.timeout(
      const Duration(milliseconds: 1800),
      onTimeout: () {
        sub.cancel();
        _isClosed = true;
        onConnectionChanged?.call(false);
        throw TimeoutException('Timeout menunggu respons router.');
      },
    );
  }

  List<int> _encodeLength(int len) {
    if (len < 0x80) return [len];
    if (len < 0x4000) return [(len >> 8) | 0x80, len & 0xFF];
    if (len < 0x200000) return [(len >> 16) | 0xC0, (len >> 8) & 0xFF, len & 0xFF];
    if (len < 0x10000000) {
      return [(len >> 24) | 0xE0, (len >> 16) & 0xFF, (len >> 8) & 0xFF, len & 0xFF];
    }
    return [0xF0, (len >> 24) & 0xFF, (len >> 16) & 0xFF, (len >> 8) & 0xFF, len & 0xFF];
  }

  Future<void> close() async {
    _isClosed = true;
    onConnectionChanged?.call(false);
    try {
      await _socket?.close();
      _socket?.destroy();
    } catch (_) {}
    _socket = null;
    if (!_controller.isClosed) await _controller.close();
  }
}

import 'dart:convert';
import 'package:shared_preferences/shared_preferences.dart';
import '../models/router_models.dart';

class ProfileStorageService {
  static const _profilesKey = 'saved_router_profiles_v1';
  static const _activeProfileIdKey = 'active_router_profile_id_v1';

  Future<List<RouterProfile>> getProfiles() async {
    final prefs = await SharedPreferences.getInstance();
    final raw = prefs.getString(_profilesKey);
    if (raw == null || raw.isEmpty) return [];

    final List<dynamic> decoded = jsonDecode(raw);
    return decoded
        .map((e) => RouterProfile.fromJson(Map<String, dynamic>.from(e)))
        .toList();
  }

  Future<void> saveProfile(RouterProfile profile, {bool setAsActive = true}) async {
    final prefs = await SharedPreferences.getInstance();
    final profiles = await getProfiles();

    final existingIdx = profiles.indexWhere((p) => p.id == profile.id);
    if (existingIdx >= 0) {
      profiles[existingIdx] = profile;
    } else {
      profiles.add(profile);
    }

    await prefs.setString(
      _profilesKey,
      jsonEncode(profiles.map((p) => p.toJson()).toList()),
    );

    if (setAsActive) {
      await prefs.setString(_activeProfileIdKey, profile.id);
    }
  }

  Future<void> deleteProfile(String id) async {
    final prefs = await SharedPreferences.getInstance();
    final profiles = await getProfiles();
    profiles.removeWhere((p) => p.id == id);

    await prefs.setString(
      _profilesKey,
      jsonEncode(profiles.map((p) => p.toJson()).toList()),
    );

    final activeId = prefs.getString(_activeProfileIdKey);
    if (activeId == id) {
      await prefs.remove(_activeProfileIdKey);
    }
  }

  Future<void> setActiveProfileId(String? id) async {
    final prefs = await SharedPreferences.getInstance();
    if (id == null) {
      await prefs.remove(_activeProfileIdKey);
    } else {
      await prefs.setString(_activeProfileIdKey, id);
    }
  }

  Future<RouterProfile?> getActiveProfile() async {
    final prefs = await SharedPreferences.getInstance();
    final activeId = prefs.getString(_activeProfileIdKey);
    if (activeId == null) return null;

    final profiles = await getProfiles();
    try {
      return profiles.firstWhere((p) => p.id == activeId);
    } catch (_) {
      return null;
    }
  }
}

import 'dart:async';
import 'dart:io';

class DiscoveredRouter {
  final String ip;
  final int port;
  final String label;

  DiscoveredRouter({
    required this.ip,
    required this.port,
    required this.label,
  });
}

class RouterDiscoveryService {
  // Port umum MikroTik: 8728 (API Binary), 80 (HTTP REST), 443 (HTTPS REST), 8729 (API SSL)
  static const List<int> _portsToProbe = [8728, 80, 443, 8729];

  /// Pindai jaringan lokal Wi-Fi untuk menemukan router MikroTik aktif
  Future<DiscoveredRouter?> discoverActiveRouter({
    Duration probeTimeout = const Duration(milliseconds: 900),
  }) async {
    final candidateIps = await _getCandidateIps();

    // Pindai kandidat gateway secara paralel
    final futures = candidateIps.map((ip) => _probeHost(ip, probeTimeout)).toList();
    final results = await Future.wait(futures);

    // Prioritaskan port 8728 (Native Binary API) jika ada beberapa port terbuka
    DiscoveredRouter? bestResult;
    for (final r in results) {
      if (r != null) {
        if (r.port == 8728) return r;
        bestResult ??= r;
      }
    }

    return bestResult;
  }

  Future<List<String>> _getCandidateIps() async {
    final candidates = <String>{};

    try {
      final interfaces = await NetworkInterface.list(
        type: InternetAddressType.IPv4,
        includeLinkLocal: false,
      );

      for (final iface in interfaces) {
        for (final addr in iface.addresses) {
          final ip = addr.address;
          if (ip == '127.0.0.1' || ip.startsWith('169.254.')) continue;

          final parts = ip.split('.');
          if (parts.length == 4) {
            final prefix = '${parts[0]}.${parts[1]}.${parts[2]}';
            // Default gateway router pada umumnya (.1 atau .254)
            candidates.add('$prefix.1');
            candidates.add('$prefix.254');
          }
        }
      }
    } catch (_) {}

    // IP Default bawaan pabrik MikroTik
    candidates.add('192.168.88.1');
    candidates.add('192.168.1.1');
    candidates.add('10.0.0.1');

    return candidates.toList();
  }

  Future<DiscoveredRouter?> _probeHost(String host, Duration timeout) async {
    for (final port in _portsToProbe) {
      try {
        final socket = await Socket.connect(host, port, timeout: timeout);
        await socket.close();
        return DiscoveredRouter(
          ip: host,
          port: port,
          label: 'Router ($host)',
        );
      } catch (_) {
        // Port tidak merespons, coba port berikutnya
      }
    }
    return null;
  }
}

class _RouterDashboardScreenState extends State<RouterDashboardScreen> {
  late MikroTikService _service;
  late Map<String, dynamic> _data;
  bool _isLoading = false;
  int _selectedTab = 0;
  bool _isOnline = true;
  Timer? _heartbeatTimer;
  StreamSubscription<bool>? _connectionSub;

  @override
  void initState() {
    super.initState();
    _service = MikroTikService(widget.profile);
    _data = widget.initialData;
    _subscribeConnection();
    _startHeartbeat();
  }

  @override
  void dispose() {
    _connectionSub?.cancel();
    _heartbeatTimer?.cancel();
    _service.dispose();
    super.dispose();
  }

  void _subscribeConnection() {
    _connectionSub =
        _service.connectionStateStream.listen((connected) {
      if (!mounted) return;
      if (_isOnline != connected) {
        setState(() => _isOnline = connected);
      }
    });
  }

  void _startHeartbeat() {
    _heartbeatTimer =
        Timer.periodic(const Duration(milliseconds: 1500), (_) async {
      if (!mounted || _isLoading) return;
      final alive = await _service.checkConnection();
      if (!mounted) return;
      if (alive != _isOnline) {
        setState(() => _isOnline = alive);
      }
      if (alive) {
        try {
          final res =
              await _service.fetchMenu('/system/resource/print');
          if (mounted && res.isNotEmpty) {
            setState(() => _data['resource'] = res.first);
          }
        } catch (_) {}
      }
    });
  }

  UserRoleInfo get _role => _data['role'] as UserRoleInfo;

  Future<void> _refresh() async {
    setState(() => _isLoading = true);
    try {
      final fresh = await _service.fetchDashboardSummary();
      if (mounted) {
        setState(() {
          _data = fresh;
          _isOnline = true;
          if (_role.isUser) _selectedTab = 0;
        });
      }
    } catch (e) {
      if (mounted) setState(() => _isOnline = false);
      _showSnack('Gagal refresh.', isError: true);
    } finally {
      if (mounted) setState(() => _isLoading = false);
    }
  }

  Future<void> _switchRouter() async {
    _heartbeatTimer?.cancel();
    await _service.dispose();
    await ProfileStorageService().setActiveProfileId(null);
    if (!mounted) return;
    Navigator.of(context).pushReplacement(
      MaterialPageRoute(builder: (_) => const ProfileListScreen()),
    );
  }

  void _showSnack(String msg, {bool isError = false}) {
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(msg, style: const TextStyle(fontSize: 13)),
        backgroundColor: isError
            ? const Color(0xFFFF3B30)
            : const Color(0xFF34C759),
      ),
    );
  }

    final activeTab = isTechnician ? _selectedTab : 0;

    return Scaffold(

              ],
            ),

              _role.isUser

                  : '${widget.profile.host} • ${_role.roleBadge}',
              style: const TextStyle(
                  fontSize: 11, color: Color(0xFF86868B)),
              maxLines: 1,
              overflow: TextOverflow.ellipsis,
            ),
          ],
        ),
        actions: [
          IconButton(
            icon: _isLoading
                ? const SizedBox(
                    width: 16,
                    height: 16,
                    child:
                        CircularProgressIndicator(strokeWidth: 2))
                : const Icon(Icons.refresh_rounded, size: 20),
            onPressed: _isLoading ? null : _refresh,
          ),
          IconButton(
            icon: const Icon(Icons.swap_horiz_rounded, size: 20),

class _CustomerTabViewState extends State<CustomerTabView> {
  String _searchQuery = '';
  final _searchController = TextEditingController();
  bool _isTestingPing = false;
  String? _pingStatus;
  Color? _pingColor;
  IconData? _pingIcon;
  final Map<String, bool> _revealedPasswords = {};
  int _deviceFilterMode = 0;

  @override
  void dispose() {
    _searchController.dispose();
    super.dispose();
  }

  Future<void> _runPingTest() async {
    setState(() {
      _isTestingPing = true;
      _pingStatus = 'Menguji koneksi...';
      _pingColor = const Color(0xFF0071E3);
      _pingIcon = Icons.sync_rounded;
    });
    try {
      final res = await widget.service.pingHost('8.8.8.8', count: 2);
      if (!mounted) return;
      int? latency;
      bool received = false;
      for (final r in res) {
        final timeStr = r['time'] ?? '';
        if (timeStr.isNotEmpty) {
          received = true;
          final match = RegExp(r'(\d+)ms').firstMatch(timeStr);
          if (match != null) latency = int.tryParse(match.group(1)!);
          break;
        }
      }
      setState(() {
        if (!received) {
          _pingStatus = 'Tidak ada internet';
          _pingColor = const Color(0xFFFF3B30);
          _pingIcon = Icons.cloud_off_rounded;
        } else if (latency != null && latency < 50) {
          _pingStatus = 'Sangat Lancar ($latency ms)';
          _pingColor = const Color(0xFF34C759);
          _pingIcon = Icons.check_circle_rounded;
        } else if (latency != null && latency < 150) {
          _pingStatus = 'Stabil ($latency ms)';
          _pingColor = const Color(0xFF0071E3);
          _pingIcon = Icons.wifi_rounded;
        } else {
          _pingStatus = 'Agak Lambat ($latency ms)';
          _pingColor = const Color(0xFFFF9500);
          _pingIcon = Icons.warning_amber_rounded;
        }
      });
    } catch (_) {
      if (mounted) {
        setState(() {
          _pingStatus = 'Gagal tes koneksi';
          _pingColor = const Color(0xFFFF3B30);
          _pingIcon = Icons.error_outline_rounded;
        });
      }
    } finally {
      if (mounted) setState(() => _isTestingPing = false);
    }
  }

  List<ConnectedDevice> get _allDevices {
    final devList = (widget.data['devices'] as List<ConnectedDevice>?) ?? [];
    if (devList.isNotEmpty) return devList;
    final staticList = (widget.data['staticDevices'] as List<StaticHostDevice>?) ?? [];
    return staticList
        .map((s) => ConnectedDevice(
              macAddress: s.macAddress,
              ipAddress: s.ipAddress,
              hostName: s.customerName,
              source: s.interfaceName,
              isBlocked: s.isBlocked,
            ))
        .toList();
  }

  List<ConnectedDevice> get _filteredDevices {
    var list = _allDevices;
    if (_deviceFilterMode == 1) list = list.where((d) => !d.isBlocked).toList();
    if (_deviceFilterMode == 2) list = list.where((d) => d.isBlocked).toList();
    if (_searchQuery.isNotEmpty) {
      final q = _searchQuery.toLowerCase();
      list = list.where((d) {
        return d.hostName.toLowerCase().contains(q) ||
            d.ipAddress.toLowerCase().contains(q) ||
            d.macAddress.toLowerCase().contains(q);
      }).toList();
    }
    return list;
  }

  Future<void> _toggleBlock(ConnectedDevice device) async {
    final willBlock = !device.isBlocked;
    final confirm = await showDialog<bool>(
      context: context,
      builder: (ctx) => AlertDialog(
        title: Text(willBlock ? 'Blokir Perangkat?' : 'Buka Blokir?'),
        content: Text(
          willBlock
              ? 'Blokir ${device.hostName} (${device.macAddress})?'
              : 'Buka blokir ${device.hostName}?',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(ctx, false),
            child: const Text('Batal'),
          ),
          FilledButton(
            style: FilledButton.styleFrom(
              backgroundColor: willBlock ? const Color(0xFFFF3B30) : const Color(0xFF34C759),
            ),
            onPressed: () => Navigator.pop(ctx, true),
            child: Text(willBlock ? 'Blokir' : 'Buka'),
          ),
        ],
      ),
    );
    if (confirm != true) return;
    try {
      await widget.service.toggleBlockMac(device.macAddress, block: willBlock, label: device.hostName);
      if (device.ipAddress.isNotEmpty && device.ipAddress != '-' && device.ipAddress != 'Terblokir') {
        await widget.service.toggleBlockIp(device.ipAddress, block: willBlock, label: device.hostName);
      }
      widget.showSnack(willBlock ? 'Diblokir.' : 'Blokir dibuka.');
      widget.onRefresh();
    } catch (e) {
      widget.showSnack('Gagal: ${e.toString()}', isError: true);
    }
  }

  void _showEditWifiDialog(WifiInterfaceInfo wifi) {
    final ssidCtrl = TextEditingController(text: wifi.ssid);
    final passCtrl = TextEditingController(text: wifi.password);

    showDialog(
      context: context,
      builder: (ctx) => AlertDialog(
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(14)),
        title: Text('Atur Wi-Fi (${wifi.name})'),
        content: Column(
          mainAxisSize: MainAxisSize.min,
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            const Text('Nama Wi-Fi (SSID)', style: TextStyle(fontSize: 12, fontWeight: FontWeight.w500, color: Color(0xFF6C6C70))),
            const SizedBox(height: 6),
            TextField(
              controller: ssidCtrl,
              decoration: InputDecoration(
                filled: true,
                fillColor: const Color(0xFFF2F2F7),
                contentPadding: const EdgeInsets.symmetric(horizontal: 12, vertical: 10),
                border: OutlineInputBorder(borderRadius: BorderRadius.circular(8), borderSide: BorderSide.none),
              ),
            ),
            const SizedBox(height: 12),
            const Text('Password Baru', style: TextStyle(fontSize: 12, fontWeight: FontWeight.w500, color: Color(0xFF6C6C70))),
            const SizedBox(height: 6),
            TextField(
              controller: passCtrl,
              decoration: InputDecoration(
                hintText: 'Min. 8 karakter',
                filled: true,
                fillColor: const Color(0xFFF2F2F7),
                contentPadding: const EdgeInsets.symmetric(horizontal: 12, vertical: 10),
                border: OutlineInputBorder(borderRadius: BorderRadius.circular(8), borderSide: BorderSide.none),
              ),
            ),
          ],
        ),
        actions: [
          TextButton(onPressed: () => Navigator.pop(ctx), child: const Text('Batal')),
          TextButton(
            onPressed: () async {
              final newSsid = ssidCtrl.text.trim();
              final newPass = passCtrl.text;
              if (newSsid.isEmpty) return;
              Navigator.pop(ctx);
              try {
                await widget.service.updateWifiConfig(
                  wifi: wifi,
                  newSsid: newSsid,
                  newPassword: newPass,
                );
                widget.showSnack('Pengaturan Wi-Fi berhasil diperbarui!');
                widget.onRefresh();
              } catch (e) {
                widget.showSnack('Gagal: $e', isError: true);
              }
            },
            child: const Text('Simpan', style: TextStyle(fontWeight: FontWeight.bold, color: Color(0xFF0071E3))),
          ),
        ],
      ),
    );
  }
