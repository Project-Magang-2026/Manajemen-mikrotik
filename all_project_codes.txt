class AutoLoginGate extends StatefulWidget {
  const AutoLoginGate({super.key});

  @override
  State<AutoLoginGate> createState() => _AutoLoginGateState();
}

class _AutoLoginGateState extends State<AutoLoginGate> {
  final _storage = ProfileStorageService();

  @override
  void initState() {
    super.initState();
    _checkSavedSession();
  }

  Future<void> _checkSavedSession() async {
    final activeProfile = await _storage.getActiveProfile();
    if (activeProfile == null) {
      _goToProfileList();
      return;
    }

    try {
      final service = MikroTikService(activeProfile);
      final data = await service.fetchDashboardSummary();
      if (!mounted) return;
      Navigator.of(context).pushReplacement(
        MaterialPageRoute(
          builder: (_) => RouterDashboardScreen(
            profile: activeProfile,
            initialData: data,
          ),
        ),
      );
    } catch (_) {
      _goToProfileList();
    }
  }

  void _goToProfileList() {
    if (!mounted) return;
    Navigator.of(context).pushReplacement(
      MaterialPageRoute(builder: (_) => const ProfileListScreen()),
    );
  }

  @override
  Widget build(BuildContext context) {
    return const Scaffold(
      backgroundColor: Color(0xFFF2F2F7),
      body: Center(
        child: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            SizedBox(
              width: 26,
              height: 26,
              child: CircularProgressIndicator(
                strokeWidth: 2.5,
                color: Color(0xFF007AFF),
              ),
            ),
            SizedBox(height: 16),
            Text(
              'Menghubungkan ke router...',
              style: TextStyle(
                fontSize: 14,
                fontWeight: FontWeight.w500,
                color: Color(0xFF8E8E93),
              ),
            ),
          ],
        ),
      ),
    );
  }
}

enum ApiProtocol {
  routerOsApi8728,
  restApiHttp,
  restApiHttps,
}

extension ApiProtocolExt on ApiProtocol {
  String get label {
    switch (this) {
      case ApiProtocol.routerOsApi8728:
        return 'RouterOS API (Port 8777)';
      case ApiProtocol.restApiHttp:
        return 'REST API HTTP (Port 80)';
      case ApiProtocol.restApiHttps:
        return 'REST API HTTPS (Port 443)';
    }
  }

  int get defaultPort {
    switch (this) {
      case ApiProtocol.routerOsApi8728:
        return 8777;
      case ApiProtocol.restApiHttp:
        return 80;
      case ApiProtocol.restApiHttps:
        return 443;
    }
  }
}

enum AppRole {
  user,
  technician,
}

enum LoginRolePreference {
  auto,
  technician,
  user,
}

extension LoginRolePreferenceExt on LoginRolePreference {
  String get label {
    switch (this) {
      case LoginRolePreference.auto:
        return 'Otomatis (Sesuai Izin Akun)';
      case LoginRolePreference.technician:
        return 'Masuk sebagai Teknisi';
      case LoginRolePreference.user:
        return 'Masuk sebagai User';
    }
  }
}

class RouterProfile {
  final String id;
  final String label;
  final String host;
  final int port;
  final String username;
  final String password;
  final ApiProtocol protocol;
  final LoginRolePreference rolePreference;

  RouterProfile({
    required this.id,
    required this.label,
    required this.host,
    required this.port,
    required this.username,
    required this.password,
    required this.protocol,
    this.rolePreference = LoginRolePreference.auto,
  });

  Map<String, dynamic> toJson() => {
        'id': id,
        'label': label,
        'host': host,
        'port': port,
        'username': username,
        'password': password,
        'protocol': protocol.name,
        'rolePreference': rolePreference.name,
      };

  factory RouterProfile.fromJson(Map<String, dynamic> json) {
    return RouterProfile(
      id: json['id'] ?? DateTime.now().millisecondsSinceEpoch.toString(),
      label: json['label'] ?? json['host'] ?? 'Router',
      host: json['host'] ?? '',
      port: json['port'] ?? 8777,
      username: json['username'] ?? '',
      password: json['password'] ?? '',
      protocol: ApiProtocol.values.firstWhere(
        (e) => e.name == json['protocol'],
        orElse: () => ApiProtocol.routerOsApi8728,
      ),
      rolePreference: LoginRolePreference.values.firstWhere(
        (e) => e.name == json['rolePreference'],
        orElse: () => LoginRolePreference.auto,
      ),
    );
  }
}

class UserRoleInfo {
  final String username;
  final String groupName;
  final List<String> policies;
  final AppRole role;
  final bool canWrite;
  final bool hasFull;
  final bool canReboot;
  final bool canManageUsers;
  final bool canRunTest;

  UserRoleInfo({
    required this.username,
    required this.groupName,
    required this.policies,
    required this.role,
    required this.canWrite,
    required this.hasFull,
    required this.canReboot,
    required this.canManageUsers,
    required this.canRunTest,
  });

  bool get isTechnician => role == AppRole.technician;
  bool get isUser => role == AppRole.user;
  bool get isReadOnly => !canWrite;

  String get roleBadge {
    if (isTechnician) {
      return hasFull ? 'TEKNISI (FULL)' : 'TEKNISI (WRITE)';
    }
    return 'MODE USER';
  }
}

import 'dart:async';
import 'dart:convert';
import 'dart:io';
import '../models/router_models.dart';

class MikroTikService {
  final RouterProfile profile;
  _RouterOsBinaryClient? _persistentBinaryClient;
  Future<void> _commandQueue = Future.value();
  bool _isDisposed = false;

  final _connectionStateController = StreamController<bool>.broadcast();
  Stream<bool> get connectionStateStream => _connectionStateController.stream;
  bool _lastKnownConnected = true;
  bool get isConnected => _lastKnownConnected;

  void _notifyConnectionState(bool connected) {
    if (_lastKnownConnected != connected) {
      _lastKnownConnected = connected;
      if (!_connectionStateController.isClosed) {
        _connectionStateController.add(connected);
      }
    }
  }

  MikroTikService(this.profile);

  /// Menutup koneksi persistent socket saat keluar atau ganti router
  Future<void> dispose() async {
    _isDisposed = true;
    await _persistentBinaryClient?.close();
    _persistentBinaryClient = null;
    if (!_connectionStateController.isClosed) {
      await _connectionStateController.close();
    }
  }

  /// Mengecek hak akses user dari /user dan /user/group di MikroTik secara serentak
  Future<UserRoleInfo> detectUserRole() async {
    final results = await Future.wait([
      _safeRunCommand('/user/print'),
      _safeRunCommand('/user/group/print'),
    ]);
    final users = results[0];
    final groups = results[1];

    final currentUser = users.firstWhere(
      (u) => u['name'] == profile.username,
      orElse: () => {'name': profile.username, 'group': 'write'},
    );

    final groupName = currentUser['group'] ?? 'read';
    final groupData = groups.firstWhere(
      (g) => g['name'] == groupName,
      orElse: () {
        final gLow = groupName.toLowerCase();
        if (gLow == 'full') return {'name': groupName, 'policy': 'full,write,read,reboot,policy,test'};
        if (gLow == 'write') return {'name': groupName, 'policy': 'write,read,reboot,test'};
        return {'name': groupName, 'policy': 'read'};
      },
    );

    final policyRaw = groupData['policy'] ?? '';
    final policies = policyRaw
        .split(',')
        .map((e) => e.trim().toLowerCase())
        .where((e) => e.isNotEmpty)
        .toList();

    if (policies.isEmpty) {
      final gLow = groupName.toLowerCase();
      if (gLow == 'full') {
        policies.addAll(['full', 'write', 'read', 'reboot', 'policy', 'test']);
      } else if (gLow == 'write') {
        policies.addAll(['write', 'read', 'reboot', 'test']);
      } else {
        policies.add('read');
      }
    }

    final hasFull = policies.contains('full');
    final canWrite = policies.contains('write') || hasFull;
    final canReboot = policies.contains('reboot') || hasFull;
    final canManageUsers = policies.contains('policy') || hasFull;
    final canRunTest = policies.contains('test') || hasFull;

    // Menentukan AppRole (User vs Teknisi) sesuai spesifikasi 2 Role
    final gNameLow = groupName.toLowerCase();
    AppRole role;

    if (profile.rolePreference == LoginRolePreference.user) {
      role = AppRole.user;
    } else if (gNameLow.contains('user') || gNameLow.contains('pelanggan') || gNameLow.contains('customer')) {
      role = AppRole.user;
    } else if (!canWrite) {
      // Jika akun router tidak memiliki izin write/full, amankan ke mode User (Read-Only)
      role = AppRole.user;
    } else {
      // Memiliki izin write/full dan tidak memilih mode user
      role = AppRole.technician;
    }

    return UserRoleInfo(
      username: profile.username,
      groupName: groupName,
      policies: policies,
      role: role,
      canWrite: canWrite,
      hasFull: hasFull,
      canReboot: canReboot,
      canManageUsers: canManageUsers,
      canRunTest: canRunTest,
    );
  }

  /// Pemeriksaan kilat apakah koneksi router aktif dan responsif (1:1 realtime heartbeat)
  Future<bool> checkConnection() async {
    try {
      final res = await runCommand('/system/identity/print');
      final ok = res.isNotEmpty;
      _notifyConnectionState(ok);
      return ok;
    } catch (_) {
      _notifyConnectionState(false);
      return false;
    }
  }

  /// Mengambil ringkasan data dashboard router secara serentak (Parallel Future.wait)
  Future<Map<String, dynamic>> fetchDashboardSummary() async {
    final roleFuture = detectUserRole();
    final identityFuture = _safeRunCommand('/system/identity/print');
    final resourceFuture = _safeRunCommand('/system/resource/print');
    final ipFuture = _safeRunCommand('/ip/address/print');
    final ifaceFuture = _safeRunCommand('/interface/print');
    final queuesFuture = _safeRunCommand('/queue/simple/print');
    final wifiFuture = fetchWifiInterfaces();
    final devicesFuture = fetchConnectedDevices();
    final staticDevicesFuture = fetchStaticDevices();

    final results = await Future.wait([
      roleFuture,
      identityFuture,
      resourceFuture,
      ipFuture,
      ifaceFuture,
      queuesFuture,
      wifiFuture,
      devicesFuture,
      staticDevicesFuture,
    ]);

    final role = results[0] as UserRoleInfo;
    final identityList = results[1] as List<Map<String, String>>;
    final resourceList = results[2] as List<Map<String, String>>;
    final ipList = results[3] as List<Map<String, String>>;
    final ifaceList = results[4] as List<Map<String, String>>;
    final queuesList = results[5] as List<Map<String, String>>;
    final wifiList = results[6] as List<WifiInterfaceInfo>;
    final devicesList = results[7] as List<ConnectedDevice>;
    final staticDevices = results[8] as List<StaticHostDevice>;

    return {
      'role': role,
      'identity': identityList.isNotEmpty ? (identityList.first['name'] ?? 'MikroTik') : 'MikroTik',
      'resource': resourceList.isNotEmpty ? resourceList.first : <String, String>{},
      'ips': ipList,
      'interfaces': ifaceList,
      'queues': queuesList,
      'wifis': wifiList,
      'devices': devicesList,
      'staticDevices': staticDevices,
    };
  }

  /// Membaca interface Wi-Fi (mendukung RouterOS v7 /interface/wifi & v6 /interface/wireless)
  Future<List<WifiInterfaceInfo>> fetchWifiInterfaces() async {
    final result = <WifiInterfaceInfo>[];

    // 1. Cek RouterOS v7 (/interface/wifi)
    final v7Wifis = await _safeRunCommand('/interface/wifi/print');
    if (v7Wifis.isNotEmpty) {
      for (final w in v7Wifis) {
        result.add(WifiInterfaceInfo(
          id: w['.id'] ?? '',
          name: w['name'] ?? 'wifi1',
          ssid: w['configuration.ssid'] ?? w['ssid'] ?? '(Belum diset)',
          password: w['security.passphrase'] ?? '',
          securityProfile: w['security'] ?? 'default',
          isRosV7Wifi: true,
          disabled: w['disabled'] == 'true',
        ));
      }
      return result;
    }

    // 2. Cek RouterOS v6 (/interface/wireless)
    final v6Wifis = await _safeRunCommand('/interface/wireless/print');
    final secProfiles = await _safeRunCommand('/interface/wireless/security-profiles/print');

    for (final w in v6Wifis) {
      final secName = w['security-profile'] ?? 'default';
      final matchedSec = secProfiles.firstWhere(
        (s) => s['name'] == secName,
        orElse: () => <String, String>{},
      );

      result.add(WifiInterfaceInfo(
        id: w['.id'] ?? '',
        name: w['name'] ?? 'wlan1',
        ssid: w['ssid'] ?? 'MikroTik',
        password: matchedSec['wpa2-pre-shared-key'] ?? matchedSec['wpa-pre-shared-key'] ?? '',
        securityProfile: secName,
        isRosV7Wifi: false,
        disabled: w['disabled'] == 'true',
      ));
    }

    return result;
  }

  /// Mengubah SSID dan Password Wi-Fi
  Future<void> updateWifiConfig({
    required WifiInterfaceInfo wifi,
    required String newSsid,
    required String newPassword,
  }) async {
    if (newPassword.isNotEmpty && newPassword.length < 8) {
      throw Exception('Password Wi-Fi minimal 8 karakter.');
    }

    if (wifi.isRosV7Wifi) {
      final params = <String, String>{
        '.id': wifi.id,
        'configuration.ssid': newSsid,
      };
      if (newPassword.isNotEmpty) {
        params['security.passphrase'] = newPassword;
        params['security.authentication-types'] = 'wpa2-psk,wpa3-psk';
      }
      await runCommand('/interface/wifi/set', params: params);
    } else {
      await runCommand('/interface/wireless/set', params: {
        '.id': wifi.id,
        'ssid': newSsid,
      });

      if (newPassword.isNotEmpty) {
        final secProfiles = await runCommand('/interface/wireless/security-profiles/print');
        final targetSec = secProfiles.firstWhere(
          (s) => s['name'] == wifi.securityProfile,
          orElse: () => <String, String>{},
        );
        if (targetSec['.id'] != null) {
          await runCommand('/interface/wireless/security-profiles/set', params: {
            '.id': targetSec['.id']!,
            'mode': 'dynamic-keys',
            'authentication-types': 'wpa2-psk',
            'wpa2-pre-shared-key': newPassword,
          });
        }
      }
    }
  }

  /// Mengambil daftar perangkat aktif (DHCP Lease + ARP + Wireless Registration) beserta status blokir
  Future<List<ConnectedDevice>> fetchConnectedDevices() async {
    final results = await Future.wait([
      _safeRunCommand('/ip/dhcp-server/lease/print'),
      _safeRunCommand('/ip/arp/print'),
      _safeRunCommand('/ip/firewall/filter/print'),
      _safeRunCommand('/interface/wireless/access-list/print'),
      _safeRunCommand('/interface/wifi/access-list/print'),
    ]);

    final leases = results[0];
    final arps = results[1];
    final firewallRules = results[2];
    final v6AccessList = results[3];
    final v7AccessList = results[4];

    final blockedMacs = <String>{};
    for (final rule in firewallRules) {
      final mac = (rule['src-mac-address'] ?? '').toUpperCase();
      if (mac.isNotEmpty && rule['action'] == 'drop' && rule['disabled'] != 'true') {
        blockedMacs.add(mac);
      }
    }
    for (final acl in v6AccessList) {
      final mac = (acl['mac-address'] ?? '').toUpperCase();
      if (mac.isNotEmpty && acl['authentication'] == 'no') {
        blockedMacs.add(mac);
      }
    }
    for (final acl in v7AccessList) {
      final mac = (acl['mac-address'] ?? '').toUpperCase();
      if (mac.isNotEmpty && acl['action'] == 'reject') {
        blockedMacs.add(mac);
      }
    }

    final deviceMap = <String, ConnectedDevice>{};

    for (final lease in leases) {
      final mac = (lease['mac-address'] ?? '').toUpperCase();
      if (mac.isEmpty) continue;
      deviceMap[mac] = ConnectedDevice(
        macAddress: mac,
        ipAddress: lease['address'] ?? lease['active-address'] ?? '-',
        hostName: (lease['host-name'] ?? '').isNotEmpty ? lease['host-name']! : 'Perangkat DHCP',
        source: 'DHCP Lease',
        isBlocked: blockedMacs.contains(mac),
      );
    }

    for (final arp in arps) {
      final mac = (arp['mac-address'] ?? '').toUpperCase();
      if (mac.isEmpty || deviceMap.containsKey(mac)) continue;
      deviceMap[mac] = ConnectedDevice(
        macAddress: mac,
        ipAddress: arp['address'] ?? '-',
        hostName: 'Host (${arp['interface'] ?? 'LAN'})',
        source: 'ARP Table',
        isBlocked: blockedMacs.contains(mac),
      );
    }

    // Pastikan perangkat yang diblokir tetap muncul meski sudah terputus dari ARP/DHCP
    for (final mac in blockedMacs) {
      if (!deviceMap.containsKey(mac)) {
        deviceMap[mac] = ConnectedDevice(
          macAddress: mac,
          ipAddress: 'Terblokir',
          hostName: 'Perangkat Diblokir',
          source: 'Firewall / ACL',
          isBlocked: true,
        );
      }
    }

    return deviceMap.values.toList();
  }

  /// Engine Static IP ISP: Menggabungkan data dari /ip/arp, /queue/simple, dan /ip/firewall/filter
  Future<List<StaticHostDevice>> fetchStaticDevices() async {
    final results = await Future.wait([
      _safeRunCommand('/ip/arp/print'),
      _safeRunCommand('/queue/simple/print'),
      _safeRunCommand('/ip/firewall/filter/print'),
    ]);

    final arpList = results[0];
    final queueList = results[1];
    final filterRules = results[2];

    // Filter drop rule yang diblokir oleh aplikasi ISP
    final blockedIps = <String>{};
    final blockedMacs = <String>{};
    for (final r in filterRules) {
      final action = r['action'] ?? '';
      final comment = r['comment'] ?? '';
      if (action == 'drop' && comment.startsWith('BLOCKED_BY_ISP_APP')) {
        final srcIp = (r['src-address'] ?? '').split('/')[0].trim();
        if (srcIp.isNotEmpty) blockedIps.add(srcIp);

        final srcMac = (r['src-mac-address'] ?? '').trim().toUpperCase();
        if (srcMac.isNotEmpty) blockedMacs.add(srcMac);
      }
    }

    // Index simple queue berdasarkan target IP (menghilangkan prefix /32)
    final queueByIp = <String, Map<String, String>>{};
    for (final q in queueList) {
      final target = q['target'] ?? '';
      final cleanTarget = target.split('/')[0].trim();
      if (cleanTarget.isNotEmpty) {
        queueByIp[cleanTarget] = q;
      }
    }

    final result = <StaticHostDevice>[];
    final seenIps = <String>{};
    final seenMacs = <String>{};

    // 1. Host aktif dari ARP table
    for (final arp in arpList) {
      final ip = (arp['address'] ?? '').trim();
      if (ip.isEmpty) continue;
      seenIps.add(ip);

      final mac = (arp['mac-address'] ?? '').trim().toUpperCase();
      if (mac.isNotEmpty) seenMacs.add(mac);

      final iface = arp['interface'] ?? 'ether';
      final arpComment = arp['comment'] ?? '';

      final q = queueByIp[ip];
      String customerName = '';
      String maxLimit = '-';
      String currentRate = '-';

      if (q != null) {
        final qComment = q['comment'] ?? '';
        final qName = q['name'] ?? '';
        if (qComment.isNotEmpty) {
          customerName = qComment;
        } else if (qName.isNotEmpty) {
          customerName = qName;
        }

        maxLimit = q['max-limit'] ?? '-';
        currentRate = q['rate'] ?? '-';
      }

      if (customerName.isEmpty && arpComment.isNotEmpty) {
        customerName = arpComment;
      }
      if (customerName.isEmpty) {
        customerName = 'Host $iface';
      }

      final isBlockedByIp = blockedIps.contains(ip);
      final isBlockedByMac = mac.isNotEmpty && blockedMacs.contains(mac);
      final isUnqueuedAnomaly = (q == null);

      result.add(StaticHostDevice(
        ipAddress: ip,
        macAddress: mac,
        interfaceName: iface,
        customerName: customerName,
        maxLimit: _formatRate(maxLimit),
        currentRate: _formatRate(currentRate),
        isBlockedByIp: isBlockedByIp,
        isBlockedByMac: isBlockedByMac,
        isUnqueuedAnomaly: isUnqueuedAnomaly,
        comment: arpComment.isNotEmpty ? arpComment : (q?['comment']),
      ));
    }

    // 2. IP yang terblokir namun sudah offline / hilang dari ARP
    for (final bIp in blockedIps) {
      if (!seenIps.contains(bIp)) {
        final q = queueByIp[bIp];
        final name = q?['comment']?.isNotEmpty == true
            ? q!['comment']!
            : (q?['name']?.isNotEmpty == true ? q!['name']! : 'Host Terblokir IP');
        result.add(StaticHostDevice(
          ipAddress: bIp,
          macAddress: '',
          interfaceName: 'Drop Rules',
          customerName: '$name (Offline)',
          maxLimit: _formatRate(q?['max-limit'] ?? '-'),
          currentRate: '-',
          isBlockedByIp: true,
          isBlockedByMac: false,
          isUnqueuedAnomaly: q == null,
          comment: 'Terblokir IP (Tidak ada di ARP)',
        ));
      }
    }

    // 3. MAC yang terblokir namun sudah offline / hilang dari ARP
    for (final bMac in blockedMacs) {
      if (!seenMacs.contains(bMac)) {
        result.add(StaticHostDevice(
          ipAddress: '-',
          macAddress: bMac,
          interfaceName: 'Drop Rules',
          customerName: 'Host Terblokir MAC (Offline)',
          maxLimit: '-',
          currentRate: '-',
          isBlockedByIp: false,
          isBlockedByMac: true,
          isUnqueuedAnomaly: false,
          comment: 'Terblokir MAC (Tidak ada di ARP)',
        ));
      }
    }

    return result;
  }

  static String _formatRate(String raw) {
    if (raw == '-' || raw.isEmpty) return '-';
    final parts = raw.split('/');
    if (parts.length == 2) {
      return '${_formatSingleRate(parts[0])} / ${_formatSingleRate(parts[1])}';
    }
    return _formatSingleRate(raw);
  }

  static String _formatSingleRate(String str) {
    final trimmed = str.trim();
    if (trimmed.isEmpty || trimmed == '0') return '0 bps';
    final val = int.tryParse(trimmed);
    if (val == null) return trimmed;
    if (val >= 1000000000) return '${(val / 1000000000).toStringAsFixed(1)} Gbps';
    if (val >= 1000000) return '${(val / 1000000).toStringAsFixed(1)} Mbps';
    if (val >= 1000) return '${(val / 1000).toStringAsFixed(0)} Kbps';
    return '$val bps';
  }

  /// Eksekutor antrean perintah FIFO persisten untuk Binary API (8728) dan REST API (80/443)
  Future<List<Map<String, String>>> runCommand(
    String path, {
    Map<String, String>? params,
  }) {
    if (_isDisposed) {
      return Future.error(Exception('MikroTikService sudah di-dispose.'));
    }

    final completer = Completer<List<Map<String, String>>>();
    _commandQueue = _commandQueue.then((_) async {
      try {
        final result = await _executeCommand(path, params: params);
        if (!completer.isCompleted) completer.complete(result);
      } catch (e, st) {
        if (!completer.isCompleted) completer.completeError(e, st);
      }
    });
    return completer.future;
  }

  Future<List<Map<String, String>>> _executeCommand(
    String path, {
    Map<String, String>? params,
  }) async {
    if (profile.protocol == ApiProtocol.routerOsApi8728) {
      if (_persistentBinaryClient == null || _persistentBinaryClient!.isClosed) {
        final client = _RouterOsBinaryClient(
          profile.host,
          profile.port,
          onConnectionChanged: _notifyConnectionState,
        );
        await client.connectAndLogin(profile.username, profile.password);
        _persistentBinaryClient = client;
        _notifyConnectionState(true);
      }

      try {
        final words = <String>[path];
        if (params != null) {
          params.forEach((k, v) => words.add('=$k=$v'));
        }
        final res = await _persistentBinaryClient!.sendCommand(words);
        _notifyConnectionState(true);
        return res;
      } catch (e) {
        // Socket terputus atau timeout, tutup client agar perintah berikutnya melakukan auto-reconnect
        _notifyConnectionState(false);
        await _persistentBinaryClient?.close();
        _persistentBinaryClient = null;
        rethrow;
      }
    } else {
      final scheme = profile.protocol == ApiProtocol.restApiHttps ? 'https' : 'http';
      final cleanPath = path.startsWith('/') ? path : '/$path';
      final url = '$scheme://${profile.host}:${profile.port}/rest$cleanPath';

      final httpClient = HttpClient()
        ..connectionTimeout = const Duration(milliseconds: 1500)
        ..badCertificateCallback = (cert, host, port) => true;

      try {
        final isPrint = cleanPath.endsWith('/print');
        final method = isPrint ? 'GET' : 'POST';
        final targetUrl = isPrint ? url.replaceFirst(RegExp(r'/print$'), '') : url;

        final request = await httpClient.openUrl(method, Uri.parse(targetUrl));
        final auth = base64Encode(utf8.encode('${profile.username}:${profile.password}'));
        request.headers.set(HttpHeaders.authorizationHeader, 'Basic $auth');
        request.headers.set(HttpHeaders.contentTypeHeader, 'application/json');

        if (!isPrint && params != null && params.isNotEmpty) {
          request.write(jsonEncode(params));
        }

        final response = await request.close();
        final body = await response.transform(utf8.decoder).join();

        if (response.statusCode == 401) {
          _notifyConnectionState(false);
          throw Exception('Username atau password salah (401).');
        }
        if (response.statusCode >= 400) {
          throw Exception('HTTP ${response.statusCode}: $body');
        }
        _notifyConnectionState(true);
        if (body.isEmpty) return [];

        final decoded = jsonDecode(body);
        if (decoded is List) {
          return decoded
              .map((item) => Map<String, String>.from(
                    (item as Map).map((k, v) => MapEntry(k.toString(), v.toString())),
                  ))
              .toList();
        } else if (decoded is Map) {
          return [
            decoded.map((k, v) => MapEntry(k.toString(), v.toString())),
          ];
        }
        return [];
      } catch (e) {
        _notifyConnectionState(false);
        rethrow;
      } finally {
        httpClient.close();
      }
    }
  }
}

/// Client socket biner MikroTik RouterOS API (Port 8728)
class _RouterOsBinaryClient {
  final String host;
  final int port;
  final void Function(bool isConnected)? onConnectionChanged;
  Socket? _socket;
  final List<int> _buffer = [];
  final List<String> _currentSentence = [];
  final StreamController<List<String>> _controller = StreamController.broadcast();
  bool _isClosed = false;

  bool get isClosed => _isClosed;

  _RouterOsBinaryClient(this.host, this.port, {this.onConnectionChanged});

  Future<void> connectAndLogin(String username, String password) async {
    _isClosed = false;
    _socket = await Socket.connect(host, port, timeout: const Duration(milliseconds: 1500));
    _socket!.setOption(SocketOption.tcpNoDelay, true);
    onConnectionChanged?.call(true);

    _socket!.listen(
      _onData,
      onError: (e) {
        _isClosed = true;
        onConnectionChanged?.call(false);
        if (!_controller.isClosed) _controller.addError(e);
      },
      onDone: () {
        _isClosed = true;
        onConnectionChanged?.call(false);
        if (!_controller.isClosed) _controller.close();
      },
      cancelOnError: false,
    );

    await sendCommand(['/login', '=name=$username', '=password=$password']);
  }

  void _onData(List<int> data) {
    _buffer.addAll(data);
    while (true) {
      final word = _decodeNextWord();
      if (word == null) break;
      if (word.isEmpty) {
        _controller.add(List<String>.from(_currentSentence));
        _currentSentence.clear();
      } else {
        _currentSentence.add(word);
      }
    }
  }

  String? _decodeNextWord() {
    if (_buffer.isEmpty) return null;
    final b0 = _buffer[0];
    int len = 0;
    int offset = 0;

    if ((b0 & 0x80) == 0x00) {
      len = b0;
      offset = 1;
    } else if ((b0 & 0xC0) == 0x80) {
      if (_buffer.length < 2) return null;
      len = ((b0 & 0x3F) << 8) | _buffer[1];
      offset = 2;
    } else if ((b0 & 0xE0) == 0xC0) {
      if (_buffer.length < 3) return null;
      len = ((b0 & 0x1F) << 16) | (_buffer[1] << 8) | _buffer[2];
      offset = 3;
    } else if ((b0 & 0xF0) == 0xE0) {
      if (_buffer.length < 4) return null;
      len = ((b0 & 0x0F) << 24) | (_buffer[1] << 16) | (_buffer[2] << 8) | _buffer[3];
      offset = 4;
    } else {
      if (_buffer.length < 5) return null;
      len = (_buffer[1] << 24) | (_buffer[2] << 16) | (_buffer[3] << 8) | _buffer[4];
      offset = 5;
    }

    if (_buffer.length < offset + len) return null;
    final bytes = _buffer.sublist(offset, offset + len);
    _buffer.removeRange(0, offset + len);
    return utf8.decode(bytes, allowMalformed: true);
  }

  Future<List<Map<String, String>>> sendCommand(List<String> words) async {
    if (_socket == null) throw Exception('Socket belum terhubung.');
    final completer = Completer<List<Map<String, String>>>();
    final records = <Map<String, String>>[];
    String? trapError;

    late StreamSubscription<List<String>> sub;
    sub = _controller.stream.listen(
      (sentence) {
        if (sentence.isEmpty) return;
        final tag = sentence.first;
        if (tag == '!re') {
          final row = <String, String>{};
          for (final w in sentence.skip(1)) {
            if (w.startsWith('=')) {
              final idx = w.indexOf('=', 1);
              if (idx != -1) row[w.substring(1, idx)] = w.substring(idx + 1);
            }
          }
          records.add(row);
        } else if (tag == '!trap') {
          for (final w in sentence.skip(1)) {
            if (w.startsWith('=message=')) {
              trapError = w.substring('=message='.length);
            }
          }
        } else if (tag == '!done') {
          sub.cancel();
          if (trapError != null) {
            completer.completeError(Exception(trapError));
          } else {
            completer.complete(records);
          }
        } else if (tag == '!fatal') {
          sub.cancel();
          completer.completeError(Exception('Koneksi diputus router (!fatal).'));
        }
      },
      onError: (e) {
        sub.cancel();
        if (!completer.isCompleted) completer.completeError(e);
      },
    );

    for (final w in words) {
      final bytes = utf8.encode(w);
      _socket!.add(_encodeLength(bytes.length));
      _socket!.add(bytes);
    }
    _socket!.add([0x00]);
    await _socket!.flush();

    return completer.future.timeout(
      const Duration(milliseconds: 1800),
      onTimeout: () {
        sub.cancel();
        _isClosed = true;
        onConnectionChanged?.call(false);
        throw TimeoutException('Timeout menunggu respons router.');
      },
    );
  }

  List<int> _encodeLength(int len) {
    if (len < 0x80) return [len];
    if (len < 0x4000) return [(len >> 8) | 0x80, len & 0xFF];
    if (len < 0x200000) return [(len >> 16) | 0xC0, (len >> 8) & 0xFF, len & 0xFF];
    if (len < 0x10000000) {
      return [(len >> 24) | 0xE0, (len >> 16) & 0xFF, (len >> 8) & 0xFF, len & 0xFF];
    }
    return [0xF0, (len >> 24) & 0xFF, (len >> 16) & 0xFF, (len >> 8) & 0xFF, len & 0xFF];
  }

  Future<void> close() async {
    _isClosed = true;
    onConnectionChanged?.call(false);
    try {
      await _socket?.close();
      _socket?.destroy();
    } catch (_) {}
    _socket = null;
    if (!_controller.isClosed) await _controller.close();
  }
}

import 'dart:convert';
import 'package:shared_preferences/shared_preferences.dart';
import '../models/router_models.dart';

class ProfileStorageService {
  static const _profilesKey = 'saved_router_profiles_v1';
  static const _activeProfileIdKey = 'active_router_profile_id_v1';

  Future<List<RouterProfile>> getProfiles() async {
    final prefs = await SharedPreferences.getInstance();
    final raw = prefs.getString(_profilesKey);
    if (raw == null || raw.isEmpty) return [];

    final List<dynamic> decoded = jsonDecode(raw);
    return decoded
        .map((e) => RouterProfile.fromJson(Map<String, dynamic>.from(e)))
        .toList();
  }

  Future<void> saveProfile(RouterProfile profile, {bool setAsActive = true}) async {
    final prefs = await SharedPreferences.getInstance();
    final profiles = await getProfiles();

    final existingIdx = profiles.indexWhere((p) => p.id == profile.id);
    if (existingIdx >= 0) {
      profiles[existingIdx] = profile;
    } else {
      profiles.add(profile);
    }

    await prefs.setString(
      _profilesKey,
      jsonEncode(profiles.map((p) => p.toJson()).toList()),
    );

    if (setAsActive) {
      await prefs.setString(_activeProfileIdKey, profile.id);
    }
  }

  Future<void> deleteProfile(String id) async {
    final prefs = await SharedPreferences.getInstance();
    final profiles = await getProfiles();
    profiles.removeWhere((p) => p.id == id);

    await prefs.setString(
      _profilesKey,
      jsonEncode(profiles.map((p) => p.toJson()).toList()),
    );

    final activeId = prefs.getString(_activeProfileIdKey);
    if (activeId == id) {
      await prefs.remove(_activeProfileIdKey);
    }
  }

  Future<void> setActiveProfileId(String? id) async {
    final prefs = await SharedPreferences.getInstance();
    if (id == null) {
      await prefs.remove(_activeProfileIdKey);
    } else {
      await prefs.setString(_activeProfileIdKey, id);
    }
  }

  Future<RouterProfile?> getActiveProfile() async {
    final prefs = await SharedPreferences.getInstance();
    final activeId = prefs.getString(_activeProfileIdKey);
    if (activeId == null) return null;

    final profiles = await getProfiles();
    try {
      return profiles.firstWhere((p) => p.id == activeId);
    } catch (_) {
      return null;
    }
  }
}

}

class _ProfileListScreenState extends State<ProfileListScreen> {
  final _storage = ProfileStorageService();
  final _discoveryService = RouterDiscoveryService();

  final _labelController = TextEditingController();
  final _ipController = TextEditingController();
  final _portController = TextEditingController();
  final _userController = TextEditingController();
  final _passController = TextEditingController();

  List<RouterProfile> _savedProfiles = [];
  bool _isConnecting = false;
  bool _isScanning = false;
  bool _obscurePass = true;
  LoginRolePreference _selectedRolePreference = LoginRolePreference.auto;

  @override
  void initState() {
    super.initState();
    _loadSavedProfiles();
  }

  Future<void> _loadSavedProfiles() async {
    final list = await _storage.getProfiles();
    if (mounted) {
      setState(() => _savedProfiles = list);
    }
  }

  Future<void> _scanNetwork() async {
    setState(() => _isScanning = true);
    try {
      final found = await _discoveryService.discoverActiveRouter();
      if (!mounted) return;
      if (found != null) {
        setState(() {
          _ipController.text = found.ip;
          _portController.text = found.port.toString();
          if (_labelController.text.trim().isEmpty) {
            _labelController.text = found.label;
          }
        });
        _showSnack('Router terdeteksi: ${found.ip}:${found.port}');
      } else {
        _showSnack('Tidak ada router/port API aktif yang terdeteksi di subnet ini.', isError: true);
      }
    } catch (e) {
      _showSnack('Gagal memindai: ${e.toString().replaceFirst("Exception: ", "")}', isError: true);
    } finally {
      if (mounted) setState(() => _isScanning = false);
    }
  }

  Future<void> _connectAndSave(RouterProfile profile) async {
    setState(() => _isConnecting = true);
    try {
      final service = MikroTikService(profile);
      final data = await service.fetchDashboardSummary();
      await _storage.saveProfile(profile, setAsActive: true);

      if (!mounted) return;
      Navigator.of(context).pushReplacement(
        MaterialPageRoute(
          builder: (_) => RouterDashboardScreen(
            profile: profile,
            initialData: data,
          ),
        ),
      );
    } catch (e) {
      _showSnack(
        'Gagal terhubung: ${e.toString().replaceFirst("Exception: ", "")}',
        isError: true,
      );
    } finally {
      if (mounted) setState(() => _isConnecting = false);
    }
  }

  void _submitNewConnection() {
    final host = _ipController.text.trim();
    final portText = _portController.text.trim();
    final port = int.tryParse(portText) ?? 8777;
    final user = _userController.text.trim();
    final pass = _passController.text;
    final label = _labelController.text.trim().isEmpty ? host : _labelController.text.trim();

    if (host.isEmpty || user.isEmpty) {
      _showSnack('IP Router dan Username wajib diisi.', isError: true);
      return;
    }

    final protocol = port == 443
        ? ApiProtocol.restApiHttps
        : port == 80
            ? ApiProtocol.restApiHttp
            : ApiProtocol.routerOsApi8728;

    final newProfile = RouterProfile(
      id: '${host}_${user}_$port',
      label: label,
      host: host,
      port: port,
      username: user,
      password: pass,
      protocol: protocol,
      rolePreference: _selectedRolePreference,
    );

    _connectAndSave(newProfile);
  }

  Future<void> _deleteProfile(RouterProfile profile) async {
    final confirm = await showDialog<bool>(
      context: context,
      builder: (ctx) => AlertDialog(
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(14)),
        title: const Text('Hapus Profil Router?'),
        content: Text('Hapus profil "${profile.label}" (${profile.host}) dari daftar tersimpan?'),
        actions: [
          TextButton(onPressed: () => Navigator.pop(ctx, false), child: const Text('Batal')),
          ElevatedButton(
            style: ElevatedButton.styleFrom(
              backgroundColor: const Color(0xFFFF3B30),
              foregroundColor: Colors.white,
              elevation: 0,
            ),
            onPressed: () => Navigator.pop(ctx, true),
            child: const Text('Hapus'),
          ),
        ],
      ),
    );
    if (confirm != true) return;

    await _storage.deleteProfile(profile.id);
    await _loadSavedProfiles();
    _showSnack('Profil "${profile.label}" berhasil dihapus.');
  }

class RouterDashboardScreen extends StatefulWidget {
  final RouterProfile profile;
  final Map<String, dynamic> initialData;

  const RouterDashboardScreen({
    super.key,
    required this.profile,
    required this.initialData,
  });

  @override
  State<RouterDashboardScreen> createState() => _RouterDashboardScreenState();
}

class _RouterDashboardScreenState extends State<RouterDashboardScreen> {
  late MikroTikService _service;
  late Map<String, dynamic> _data;
  bool _isLoading = false;
  int _selectedTab = 0; // 0: Pelanggan, 1: Winbox PC

  bool _isOnline = true;
  Timer? _liveHeartbeatTimer;
  StreamSubscription<bool>? _connectionSub;

  @override
  void initState() {
    super.initState();
    _service = MikroTikService(widget.profile);
    _data = widget.initialData;
    _subscribeConnectionStream();
    _startLiveHeartbeat();
  }

  @override
  void dispose() {
    _connectionSub?.cancel();
    _liveHeartbeatTimer?.cancel();
    _service.dispose();
    super.dispose();
  }

  void _subscribeConnectionStream() {
    _connectionSub?.cancel();
    _connectionSub = _service.connectionStateStream.listen((isConnected) {
      if (!mounted) return;
      if (_isOnline != isConnected) {
        setState(() {
          _isOnline = isConnected;
          if (isConnected) {
            _showSnack('Koneksi router kembali aktif (Online).');
          } else {
            _showSnack('Koneksi ke router terputus (Offline)!', isError: true);
          }
        });
      }
    });
  }

  void _startLiveHeartbeat() {
    _liveHeartbeatTimer?.cancel();
    // Heartbeat ringan setiap 1200ms untuk deteksi instan jika terjadi silent drop / kabel putus
    _liveHeartbeatTimer = Timer.periodic(const Duration(milliseconds: 1200), (_) async {
      if (!mounted || _isLoading) return;
      final isAlive = await _service.checkConnection();
      if (!mounted) return;

      if (isAlive != _isOnline) {
        setState(() {
          _isOnline = isAlive;
          if (isAlive) {
            _showSnack('Koneksi router kembali aktif (Online).');
          } else {
            _showSnack('Koneksi ke router terputus (Offline)!', isError: true);
          }
        });
      }

      // Update ringan status resource (CPU, Uptime) di background tanpa loading spinner
      if (isAlive) {
        try {
          final res = await _service.fetchMenu('/system/resource/print');
          if (mounted && res.isNotEmpty) {
            setState(() {
              _data['resource'] = res.first;
            });
          }
        } catch (_) {}
      }
    });
  }

  UserRoleInfo get _role => _data['role'] as UserRoleInfo;

  Future<void> _refresh() async {
    setState(() => _isLoading = true);
    try {
      final fresh = await _service.fetchDashboardSummary();
      if (mounted) {
        setState(() {
          _data = fresh;
          _isOnline = true;
          if (_role.isUser) {
            _selectedTab = 0;
          }
        });
      }
    } catch (e) {
      if (mounted) {
        setState(() => _isOnline = false);
      }
      _showSnack('Gagal memuat ulang: ${e.toString().replaceFirst("Exception: ", "")}', isError: true);
    } finally {
      if (mounted) setState(() => _isLoading = false);
    }
  }

  Future<void> _switchRouter() async {
    _liveHeartbeatTimer?.cancel();
    await _service.dispose();
    await ProfileStorageService().setActiveProfileId(null);
    if (!mounted) return;
    Navigator.of(context).pushReplacement(
      MaterialPageRoute(builder: (_) => const ProfileListScreen()),
    );
  }

  void _showSnack(String msg, {bool isError = false}) {
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(msg),
        behavior: SnackBarBehavior.floating,
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(10)),
        backgroundColor: isError ? const Color(0xFFFF3B30) : const Color(0xFF34C759),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    // Mode User dikunci hanya ke tab Ringkasan & Perangkat
    final showSwitcher = _role.isTechnician;
    final activeTab = _role.isTechnician ? _selectedTab : 0;

    return Scaffold(
      backgroundColor: const Color(0xFFF2F2F7),
      appBar: AppBar(
        titleSpacing: 16,
        title: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          mainAxisSize: MainAxisSize.min,
          children: [
            Row(
              children: [
                Flexible(
                  child: Text(
                    _data['identity']?.toString() ?? widget.profile.label,
                    style: const TextStyle(fontSize: 16, fontWeight: FontWeight.bold, color: Color(0xFF1C1C1E)),
                    maxLines: 1,
                    overflow: TextOverflow.ellipsis,
                  ),
                ),
                const SizedBox(width: 6),

  Future<void> _runConnectionTest() async {
    setState(() {
      _isTestingPing = true;
      _pingStatusText = 'Menguji latensi ke internet...';
      _pingStatusColor = const Color(0xFF007AFF);
      _pingStatusIcon = Icons.sync_rounded;
    });

    try {
      final res = await widget.service.pingHost('8.8.8.8', count: 2);
      if (!mounted) return;

      int? latencyMs;
      bool receivedReply = false;

      for (final r in res) {
        final timeStr = r['time'] ?? '';
        if (timeStr.isNotEmpty) {
          receivedReply = true;
          // Format e.g. "15ms", "15ms20us", or "0.015s"
          final match = RegExp(r'(\d+)ms').firstMatch(timeStr);
          if (match != null) {
            latencyMs = int.tryParse(match.group(1)!);
          } else {
            final numOnly = RegExp(r'(\d+)').firstMatch(timeStr);
            if (numOnly != null) latencyMs = int.tryParse(numOnly.group(1)!);
          }
          break;
        }
      }

      setState(() {
        if (!receivedReply) {
          _pingStatusText = 'Tidak Terhubung / Tidak Ada Internet';
          _pingStatusColor = const Color(0xFFFF3B30);
          _pingStatusIcon = Icons.cloud_off_rounded;
        } else if (latencyMs != null && latencyMs < 50) {
          _pingStatusText = 'Koneksi Sangat Lancar ($latencyMs ms)';
          _pingStatusColor = const Color(0xFF34C759);
          _pingStatusIcon = Icons.check_circle_rounded;
        } else if (latencyMs != null && latencyMs < 150) {
          _pingStatusText = 'Koneksi Stabil ($latencyMs ms)';
          _pingStatusColor = const Color(0xFF007AFF);
          _pingStatusIcon = Icons.wifi_rounded;
        } else if (latencyMs != null) {
          _pingStatusText = 'Koneksi Agak Lambat ($latencyMs ms)';
          _pingStatusColor = const Color(0xFFFF9500);
          _pingStatusIcon = Icons.warning_amber_rounded;
        } else {
          _pingStatusText = 'Koneksi Terhubung Normal';
          _pingStatusColor = const Color(0xFF34C759);
          _pingStatusIcon = Icons.check_circle_rounded;
        }
      });
    } catch (_) {
      if (mounted) {
        setState(() {
          _pingStatusText = 'Gagal memeriksa koneksi';
          _pingStatusColor = const Color(0xFFFF3B30);
          _pingStatusIcon = Icons.error_outline_rounded;
        });
      }
    } finally {
      if (mounted) setState(() => _isTestingPing = false);
    }
  }

  Future<void> _handleToggleBlock(ConnectedDevice device) async {
    final willBlock = !device.isBlocked;
    final actionText = willBlock ? 'Blokir' : 'Buka Blokir';

    final confirm = await showDialog<bool>(
      context: context,
      builder: (ctx) => AlertDialog(
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(16)),
        title: Text('$actionText Perangkat?', style: const TextStyle(fontWeight: FontWeight.bold, fontSize: 17)),
        content: Text(
          willBlock
              ? 'Blokir akses internet untuk ${device.hostName} (${device.ipAddress} • ${device.macAddress})?'
              : 'Buka kembali akses internet untuk ${device.hostName} (${device.macAddress})?',
          style: const TextStyle(fontSize: 14, color: Color(0xFF3C3C43)),
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.of(ctx).pop(false),
            child: const Text('Batal', style: TextStyle(color: Color(0xFF8E8E93))),
          ),
          ElevatedButton(
            style: ElevatedButton.styleFrom(
              backgroundColor: willBlock ? const Color(0xFFFF3B30) : const Color(0xFF34C759),
              foregroundColor: Colors.white,
              elevation: 0,
              shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(8)),
            ),
            onPressed: () => Navigator.of(ctx).pop(true),
            child: Text(actionText, style: const TextStyle(fontWeight: FontWeight.w600)),
          ),
        ],
      ),
    );

    if (confirm != true) return;

    try {
      await widget.service.toggleBlockMac(device.macAddress, block: willBlock, label: device.hostName);
      if (device.ipAddress.isNotEmpty && device.ipAddress != '-' && device.ipAddress != 'Terblokir') {
        await widget.service.toggleBlockIp(device.ipAddress, block: willBlock, label: device.hostName);
      }
      widget.showSnack(willBlock ? 'Perangkat ${device.hostName} berhasil diblokir.' : 'Blokir untuk ${device.hostName} dibuka.');
      widget.onRefresh();
    } catch (e) {
      widget.showSnack('Aksi ditolak router: ${e.toString().replaceFirst("Exception: ", "")}', isError: true);
    }
  }

  List<ConnectedDevice> get _allDevices {
    final devList = (widget.data['devices'] as List<ConnectedDevice>?) ?? [];
    if (devList.isNotEmpty) return devList;

    // Fallback dari staticDevices jika devices kosong
    final staticList = (widget.data['staticDevices'] as List<StaticHostDevice>?) ?? [];
    return staticList.map((s) => ConnectedDevice(
      macAddress: s.macAddress,
      ipAddress: s.ipAddress,
      hostName: s.customerName,
      source: s.interfaceName,
      isBlocked: s.isBlocked,
    )).toList();
  }

  List<ConnectedDevice> get _filteredDevices {
    final query = _searchQuery.trim().toLowerCase();
    var list = _allDevices;
    if (_deviceFilterMode == 1) {
      list = list.where((d) => !d.isBlocked).toList();
    } else if (_deviceFilterMode == 2) {
      list = list.where((d) => d.isBlocked).toList();
    }
    if (query.isEmpty) return list;
    return list.where((d) {
      final name = d.hostName.toLowerCase();
      final ip = d.ipAddress.toLowerCase();
      final mac = d.macAddress.toLowerCase();
      return name.contains(query) || ip.contains(query) || mac.contains(query);
    }).toList();
  }

import 'dart:async';
import 'package:flutter/material.dart';

class WinboxPanelView extends StatefulWidget {
  final MikroTikService service;
  final Map<String, dynamic> data;
  final Function(String, {bool isError}) showSnack;

  const WinboxPanelView({
    super.key,
    required this.service,
    required this.data,
    required this.showSnack,
  });

  @override
  State<WinboxPanelView> createState() => _WinboxPanelViewState();
}

class _WinboxPanelViewState extends State<WinboxPanelView> {
  int _activeCategory = 6; // Default: IP
  int _activeSubMenu = 0;   // Default: Addresses
  String _searchQuery = '';
  bool _isSearchOpen = false;

  List<Map<String, String>> _currentMenuItems = [];
  bool _isMenuLoading = false;
  final _searchController = TextEditingController();

  bool _safeModeActive = false;
  bool _isAutoSyncLive = true;
  Timer? _liveAutoSyncTimer;

  @override
  void initState() {
    super.initState();
    _loadActiveWinboxSubMenu();
    _startLiveAutoSync();
  }

  @override
  void dispose() {
    _liveAutoSyncTimer?.cancel();
    _searchController.dispose();
    super.dispose();
  }

  void _startLiveAutoSync() {
    _liveAutoSyncTimer?.cancel();
    _liveAutoSyncTimer = Timer.periodic(const Duration(milliseconds: 2000), (_) {
      if (!mounted || !_isAutoSyncLive || _isMenuLoading || _isSearchOpen || _isSpecialView) return;
      _loadActiveWinboxSubMenu(silent: true);
    });
  }

  WinboxCategory get _activeCat => winboxCategories[_activeCategory];
  WinboxSubMenu get _activeSub => _activeCat.subMenus[_activeSubMenu];

  bool get _isSpecialView {
    final p = _activeSub.path;
    return p == 'PING_TOOL' ||
        p == 'TRACEROUTE_TOOL' ||
        p == 'PROFILE_TOOL' ||
        p == 'CLI_TOOL' ||
        p == 'SYSTEM_REBOOT' ||
        p == 'SYSTEM_SHUTDOWN';
  }

  bool get _isAddable {
    final p = _activeSub.path;
    if (_isSpecialView) return false;
    return p != '/log' &&
        p != '/system/resource' &&
        p != '/system/routerboard' &&
        p != '/system/health' &&
        p != '/user/active';
  }

  Future<void> _loadActiveWinboxSubMenu({bool silent = false}) async {
    final sub = _activeSub;
    if (_isSpecialView) {
      if (!silent && mounted) setState(() => _isMenuLoading = false);
      return;
    }

    if (!silent && mounted) {
      setState(() => _isMenuLoading = true);
    }

    try {
      final printPath = sub.path.endsWith('/print') ? sub.path : '${sub.path}/print';
      final res = await widget.service.fetchMenu(printPath);
      if (mounted) {
        setState(() => _currentMenuItems = res);
      }
    } catch (_) {
      if (!silent && mounted) setState(() => _currentMenuItems = []);
    } finally {
      if (!silent && mounted) setState(() => _isMenuLoading = false);
    }
  }

  List<Map<String, String>> get _filteredItems {
    if (_searchQuery.isEmpty) return _currentMenuItems;
    final q = _searchQuery.toLowerCase();
    return _currentMenuItems.where((item) {
      return item.values.any((v) => v.toLowerCase().contains(q));
    }).toList();
  }

  void _onAdd() {
    final sub = _activeSub;
    final ifaces = (widget.data['interfaces'] as List<Map<String, String>>?) ?? [];

    if (sub.path == '/ip/address') {
      WinboxDialogs.showAddIpDialog(
        context: context,
        service: widget.service,
        interfaces: ifaces,
        onSuccess: _loadActiveWinboxSubMenu,
        showSnack: widget.showSnack,
      );
    } else if (sub.path == '/ip/route') {
      WinboxDialogs.showAddRouteDialog(
        context: context,
        service: widget.service,
        onSuccess: _loadActiveWinboxSubMenu,
        showSnack: widget.showSnack,
      );
    } else if (sub.path == '/queue/simple') {
      WinboxDialogs.showAddQueueDialog(
        context: context,
        service: widget.service,
        onSuccess: _loadActiveWinboxSubMenu,
        showSnack: widget.showSnack,
      );
    } else if (sub.path == '/ppp/secret') {
      WinboxDialogs.showAddPppSecretDialog(
        context: context,
        service: widget.service,
        onSuccess: _loadActiveWinboxSubMenu,
        showSnack: widget.showSnack,
      );
    } else {
      final role = widget.data['role'] as UserRoleInfo?;
      WinboxDialogs.showGenericAddDialog(
        context: context,
        service: widget.service,
        basePath: sub.path,
        canManageUsers: role?.canManageUsers ?? true,
        onSuccess: _loadActiveWinboxSubMenu,
        showSnack: widget.showSnack,
      );
    }
  }

  Future<void> _onToggleItem(Map<String, String> item, bool enable) async {
    final id = item['.id'];
    if (id == null) return;
    final name = item[_activeSub.primaryField] ?? item['name'] ?? item['address'] ?? 'Item';
    final actionText = enable ? 'mengaktifkan (enable)' : 'menonaktifkan (disable)';

    final confirm = await WinboxDialogs.showConfirmActionDialog(
      context: context,
      title: 'Konfirmasi',
      message: 'Apakah Anda yakin ingin $actionText "$name" (${_activeSub.path})?',
      confirmLabel: enable ? 'Yakin (Enable)' : 'Yakin (Disable)',
      confirmColor: enable ? const Color(0xFF34C759) : const Color(0xFFFF9500),
    );
    if (!confirm) return;

    try {
      final cmd = enable ? '${_activeSub.path}/enable' : '${_activeSub.path}/disable';
      await widget.service.executePath(cmd, params: {'.id': id});
      widget.showSnack(enable ? 'Item "$name" diaktifkan (Enabled).' : 'Item "$name" dinonaktifkan (Disabled).');
      await _loadActiveWinboxSubMenu();
    } catch (e) {
      widget.showSnack('Gagal: $e', isError: true);
    }
  }

  void _onCommentItem(Map<String, String> item) {
    final id = item['.id'];
    if (id == null) return;
    final name = item[_activeSub.primaryField] ?? item['name'] ?? item['address'] ?? 'Item';
    final commentCtrl = TextEditingController(text: item['comment'] ?? '');

    showDialog(
      context: context,
      builder: (ctx) => AlertDialog(
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(14)),
        title: Text('Winbox Comment ($name)'),
        content: TextField(
          controller: commentCtrl,
          autofocus: true,
          style: const TextStyle(fontSize: 13.5, color: Color(0xFF1C1C1E)),
          decoration: const InputDecoration(hintText: 'Tulis komentar untuk item ini...'),
        ),
        actions: [
          TextButton(onPressed: () => Navigator.pop(ctx), child: const Text('Batal')),
          ElevatedButton(
            style: ElevatedButton.styleFrom(
              backgroundColor: const Color(0xFF007AFF),
              foregroundColor: Colors.white,
              elevation: 0,
            ),
            onPressed: () async {
              final newComment = commentCtrl.text.trim();
              final oldComment = (item['comment'] ?? '').trim();
              if (newComment == oldComment) {
                Navigator.pop(ctx);
                return;
              }

              final confirm = await WinboxDialogs.showConfirmActionDialog(
                context: context,
                title: 'Konfirmasi Ubah Komentar',
                message: 'Apakah Anda sudah yakin ingin menerapkan komentar ini pada "$name"?\n\n• Komentar: "${newComment.isEmpty ? "(dikosongkan)" : newComment}"',
                confirmLabel: 'Yakin & Simpan',
              );
              if (!confirm) return;

              Navigator.pop(ctx);
              try {
                await widget.service.setComment(_activeSub.path, id, newComment);
                widget.showSnack('Komentar untuk "$name" berhasil disimpan.');
                await _loadActiveWinboxSubMenu();
              } catch (e) {
                widget.showSnack('Gagal: $e', isError: true);
              }
            },
            child: const Text('Simpan'),
          ),
        ],
      ),
    );
  }

  Future<void> _onRemoveItem(Map<String, String> item) async {
    final id = item['.id'];
    if (id == null) return;
    final name = item[_activeSub.primaryField] ?? item['name'] ?? item['address'] ?? 'Item';

    final confirm = await WinboxDialogs.showConfirmActionDialog(
      context: context,
      title: 'Hapus Item (Winbox Remove)?',
      message: 'Apakah Anda yakin ingin menghapus "$name" (${_activeSub.path}) secara permanen dari router?',
      confirmLabel: 'Hapus Permanen',
      isDestructive: true,
    );

    if (confirm) {
      try {
        await widget.service.executePath('${_activeSub.path}/remove', params: {'.id': id});
        widget.showSnack('Item "$name" berhasil dihapus.');
        await _loadActiveWinboxSubMenu();
      } catch (e) {
        widget.showSnack('Gagal: $e', isError: true);
      }
    }
  }

      return TracerouteToolWidget(service: widget.service);
    } else if (sub.path == 'PROFILE_TOOL') {
      return CpuProfileToolWidget(service: widget.service);
    } else if (sub.path == 'CLI_TOOL') {
      return CliTerminalWidget(service: widget.service);
    } else if (sub.path == 'SYSTEM_REBOOT' || sub.path == 'SYSTEM_SHUTDOWN') {
      return _buildSystemActionView(sub);

    if (_isMenuLoading) {
      return const Center(child: CircularProgressIndicator(strokeWidth: 2));

  static void showGenericAddDialog({
    required BuildContext context,
    required MikroTikService service,
    required String basePath,
    required VoidCallback onSuccess,
    required Function(String, {bool isError}) showSnack,
    bool canManageUsers = true,
  }) {
    if (basePath.contains('/user') && !canManageUsers) {
      showSnack('Perubahan pada menu /user membutuhkan izin "policy" atau "full".', isError: true);
      return;
    }

    final key1Ctrl = TextEditingController();
    final val1Ctrl = TextEditingController();
    final key2Ctrl = TextEditingController();
    final val2Ctrl = TextEditingController();

    showDialog(
      context: context,
      builder: (ctx) => AlertDialog(
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(14)),
        title: Text('Tambah ke $basePath'),
        content: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            Row(
              children: [
                Expanded(child: buildDialogField(controller: key1Ctrl, label: 'Parameter 1', placeholder: 'name')),
                const SizedBox(width: 8),
                Expanded(child: buildDialogField(controller: val1Ctrl, label: 'Nilai 1', placeholder: 'nilai')),
              ],
            ),
            const SizedBox(height: 10),
            Row(
              children: [
                Expanded(child: buildDialogField(controller: key2Ctrl, label: 'Parameter 2 (Opsional)', placeholder: 'target')),
                const SizedBox(width: 8),
                Expanded(child: buildDialogField(controller: val2Ctrl, label: 'Nilai 2', placeholder: 'nilai')),
              ],
            ),
          ],
        ),
        actions: [
          TextButton(onPressed: () => Navigator.pop(ctx), child: const Text('Batal')),
          TextButton(
            onPressed: () async {
              if (key1Ctrl.text.isEmpty || val1Ctrl.text.isEmpty) return;
              final params = <String, String>{key1Ctrl.text.trim(): val1Ctrl.text.trim()};
              if (key2Ctrl.text.isNotEmpty && val2Ctrl.text.isNotEmpty) {
                params[key2Ctrl.text.trim()] = val2Ctrl.text.trim();
              }

              final confirm = await showConfirmActionDialog(
                context: context,
                title: 'Konfirmasi Tambah Item',
                message: 'Apakah Anda sudah yakin ingin menambahkan data konfigurasi baru ini ke $basePath?\n\n${params.entries.map((e) => "• ${e.key}: ${e.value}").join("\n")}',

              );
              if (!confirm) return;

              Navigator.pop(ctx);
              try {
                await service.executePath('$basePath/add', params: params);
                showSnack('Berhasil ditambahkan ke $basePath.');
                onSuccess();
              } catch (e) {
                showSnack('Gagal: $e', isError: true);
              }
            },
            child: const Text('Tambah', style: TextStyle(fontWeight: FontWeight.bold)),
          ),
        ],
      ),
    );
  }

import 'package:flutter/material.dart';
import '../services/mikrotik_service.dart';
import 'winbox_dialogs.dart';

class WinboxItemInspector {
  static void show({
    required BuildContext context,
    required Map<String, String> item,
    required String basePath,
    required MikroTikService service,
    required VoidCallback onRefresh,
    required Function(String, {bool isError}) showSnack,
  }) {
    final id = item['.id'];
    final name = item['name'] ?? item['address'] ?? item['dst-address'] ?? item['chain'] ?? 'Detail Item';
    final disabled = item['disabled'] == 'true';

    // Editable controllers map
    final controllers = <String, TextEditingController>{};
    for (final entry in item.entries) {
      if (!entry.key.startsWith('.')) {
        controllers[entry.key] = TextEditingController(text: entry.value);
      }
    }

    showModalBottomSheet(
      context: context,
      isScrollControlled: true,
      backgroundColor: Colors.white,
      shape: const RoundedRectangleBorder(
        borderRadius: BorderRadius.vertical(top: Radius.circular(16)),
      ),
      builder: (ctx) => StatefulBuilder(
        builder: (ctx, setSheetState) => DraggableScrollableSheet(
          initialChildSize: 0.75,
          maxChildSize: 0.95,
          minChildSize: 0.45,
          expand: false,
          builder: (context, scrollCtrl) => Padding(
            padding: const EdgeInsets.fromLTRB(16, 10, 16, 20),
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.stretch,
              children: [
                // Winbox Grab & Title
                Center(
                  child: Container(
                    width: 36,
                    height: 4,
                    decoration: BoxDecoration(color: const Color(0xFFD1D1D6), borderRadius: BorderRadius.circular(2)),
                  ),
                ),
                const SizedBox(height: 10),
                Row(
                  children: [
                    Container(
                      padding: const EdgeInsets.symmetric(horizontal: 6, vertical: 2),
                      decoration: BoxDecoration(
                        color: const Color(0xFF007AFF).withValues(alpha: 0.12),
                        borderRadius: BorderRadius.circular(4),
                      ),
                      child: Text(
                        basePath,
                        style: const TextStyle(fontSize: 10, fontWeight: FontWeight.bold, color: Color(0xFF007AFF), fontFamily: 'monospace'),
                      ),
                    ),
                    const SizedBox(width: 8),
                    Expanded(
                      child: Text(
                        name,
                        style: const TextStyle(fontSize: 16, fontWeight: FontWeight.bold, color: Color(0xFF1C1C1E)),
                        overflow: TextOverflow.ellipsis,
                      ),
                    ),
                  ],
                ),
                const SizedBox(height: 10),

                // WINBOX ACTION BUTTONS BAR (OK, Cancel, Apply, Remove, Enable/Disable)
                Row(
                  children: [
                    // Apply Button
                    if (id != null)
                      Expanded(
                        child: TextButton(
                          style: TextButton.styleFrom(
                            backgroundColor: const Color(0xFF007AFF),
                            foregroundColor: Colors.white,
                            padding: const EdgeInsets.symmetric(vertical: 8),
                            shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(6)),
                          ),
                          onPressed: () async {
                            final updatedProps = <String, String>{};
                            controllers.forEach((k, v) {
                              final currentVal = v.text.trim();
                              final originalVal = (item[k] ?? '').trim();
                              if (currentVal != originalVal) {
                                updatedProps[k] = currentVal;
                              }
                            });

                            if (updatedProps.isEmpty) {
                              showSnack('Tidak ada perubahan konfigurasi yang dilakukan.');
                              return;
                            }

                            final changesSummary = updatedProps.entries.map((e) {
                              final oldV = item[e.key] ?? '(kosong)';
                              return '• ${e.key}: "$oldV" ➔ "${e.value}"';
                            }).join('\n');

                            final confirm = await WinboxDialogs.showConfirmActionDialog(
                              context: context,
                              title: 'Konfirmasi Perubahan Konfigurasi',
                              message: 'Apakah Anda sudah yakin ingin menerapkan perubahan konfigurasi berikut ke MikroTik?\n\n$changesSummary',
                              confirmLabel: 'Yakin & Terapkan',
                              confirmColor: const Color(0xFF007AFF),
                            );

                            if (!confirm) return;

                            // Tutup sheet terlebih dahulu agar notifikasi hijau langsung terlihat di layar
                            Navigator.pop(ctx);

                            try {
                              await service.setItemProperties(basePath, id, updatedProps);
                              showSnack('Perubahan konfigurasi untuk "$name" berhasil diterapkan!');
                              onRefresh();
                            } catch (e) {
                              showSnack('Gagal: $e', isError: true);
                            }
                          },
                          child: const Text('Apply', style: TextStyle(fontWeight: FontWeight.bold, fontSize: 12)),
                        ),
                      ),
                    const SizedBox(width: 6),
                    // Enable/Disable
                    if (id != null && item.containsKey('disabled'))
                      Expanded(
                        child: TextButton(
                          style: TextButton.styleFrom(
                            backgroundColor: disabled ? const Color(0xFFE8F5E9) : const Color(0xFFFFF3E0),
                            foregroundColor: disabled ? const Color(0xFF2E7D32) : const Color(0xFFE65100),
                            padding: const EdgeInsets.symmetric(vertical: 8),
                            shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(6)),
                          ),
                          onPressed: () async {
                            final willEnable = disabled;
                            final actionText = willEnable ? 'mengaktifkan (enable)' : 'menonaktifkan (disable)';

                            final confirm = await WinboxDialogs.showConfirmActionDialog(
                              context: context,
                              title: 'Konfirmasi',
                              message: 'Apakah Anda yakin ingin $actionText "$name" ($basePath)?',
                              confirmLabel: willEnable ? 'Yakin (Enable)' : 'Yakin (Disable)',
                              confirmColor: willEnable ? const Color(0xFF34C759) : const Color(0xFFFF9500),
                            );

                            if (!confirm) return;

                            Navigator.pop(ctx);
                            try {
                              final cmd = willEnable ? '$basePath/enable' : '$basePath/disable';

  const CliTerminalWidget({super.key, required this.service});

  @override

}

class _CliTerminalWidgetState extends State<CliTerminalWidget> {
  final _cliPathController = TextEditingController(text: '/ip/route/print');
  bool _isLoading = false;
  String _output = '';

  final List<String> _quickCommands = [
    '/system/resource/print',
    '/ip/address/print',
    '/interface/print',
    '/ip/route/print',
    '/ip/dhcp-server/lease/print',
    '/log/print',
    '/ip/firewall/filter/print',
  ];

  @override
  void dispose() {
    _cliPathController.dispose();
    super.dispose();
  }

  Future<void> _runCommand([String? overrideCmd]) async {
    final cmd = (overrideCmd ?? _cliPathController.text).trim();
    if (cmd.isEmpty) return;

    if (overrideCmd != null) {
      _cliPathController.text = overrideCmd;
    }

    setState(() => _isLoading = true);
    try {
      String path = cmd;
      Map<String, String>? params;

      if (cmd.contains(' ')) {
        final parts = cmd.split(RegExp(r'\s+'));
        if (parts.length > 1 && !parts[1].contains('=')) {
          if (parts.every((p) => !p.contains('='))) {
            path = parts.map((p) => p.replaceAll('/', '')).where((p) => p.isNotEmpty).join('/');
            path = '/$path';
          } else {
            path = parts[0];
            params = {};
            for (int i = 1; i < parts.length; i++) {
              final kv = parts[i].split('=');
              if (kv.length == 2) {
                params[kv[0]] = kv[1];
              } else if (kv.length == 1) {
                params['address'] = kv[0];
              }
            }
          }
        } else {
          path = parts[0];
          params = {};
          for (int i = 1; i < parts.length; i++) {
            final kv = parts[i].split('=');
            if (kv.length == 2) {
              params[kv[0]] = kv[1];
            }
          }
        }
      }

      final res = await widget.service.runCommand(path, params: params);
      setState(() {
        _output = res.isEmpty
            ? '(Perintah "$path" berhasil dieksekusi tanpa baris kembalian)'
            : res.map((row) => row.entries.map((e) => '${e.key}=${e.value}').join('  ')).join('\n\n');
      });
    } catch (e) {
      setState(() => _output = 'ERROR: ${e.toString().replaceFirst("Exception: ", "")}');
    } finally {
      if (mounted) setState(() => _isLoading = false);
    }
  }

  void _copyOutput() {
    if (_output.isEmpty) return;
    Clipboard.setData(ClipboardData(text: _output));
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: const Text('Output terminal berhasil disalin.'),
        behavior: SnackBarBehavior.floating,
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(10)),
        backgroundColor: const Color(0xFF34C759),
        duration: const Duration(seconds: 2),
      ),
    );
  }

    setState(() => _output = '');
  }
