import 'dart:async';
import 'package:flutter/material.dart';
import '../models/router_models.dart';
import '../models/winbox_models.dart';
import '../services/mikrotik_service.dart';
import 'winbox_dialogs.dart';
import 'winbox_item_inspector.dart';
import 'winbox_tools_view.dart';

class WinboxPanelView extends StatefulWidget {
  final MikroTikService service;
  final Map<String, dynamic> data;
  final Function(String, {bool isError}) showSnack;

  const WinboxPanelView({
    super.key,
    required this.service,
    required this.data,
    required this.showSnack,
  });

  @override
  State<WinboxPanelView> createState() => _WinboxPanelViewState();
}

class _WinboxPanelViewState extends State<WinboxPanelView> {
  int _activeCategory = 6; // Default: IP
  int _activeSubMenu = 0;   // Default: Addresses
  String _searchQuery = '';
  bool _isSearchOpen = false;

  List<Map<String, String>> _currentMenuItems = [];
  bool _isMenuLoading = false;
  final _searchController = TextEditingController();

  bool _safeModeActive = false;
  bool _isAutoSyncLive = true;
  Timer? _liveAutoSyncTimer;

  @override
  void initState() {
    super.initState();
    _loadActiveWinboxSubMenu();
    _startLiveAutoSync();
  }

  @override
  void dispose() {
    _liveAutoSyncTimer?.cancel();
    _searchController.dispose();
    super.dispose();
  }

  void _startLiveAutoSync() {
    _liveAutoSyncTimer?.cancel();
    _liveAutoSyncTimer = Timer.periodic(const Duration(milliseconds: 2000), (_) {
      if (!mounted || !_isAutoSyncLive || _isMenuLoading || _isSearchOpen || _isSpecialView) return;
      _loadActiveWinboxSubMenu(silent: true);
    });
  }

  WinboxCategory get _activeCat => winboxCategories[_activeCategory];
  WinboxSubMenu get _activeSub => _activeCat.subMenus[_activeSubMenu];

  bool get _isSpecialView {
    final p = _activeSub.path;
    return p == 'PING_TOOL' ||
        p == 'TRACEROUTE_TOOL' ||
        p == 'PROFILE_TOOL' ||
        p == 'CLI_TOOL' ||
        p == 'SYSTEM_REBOOT' ||
        p == 'SYSTEM_SHUTDOWN';
  }

  bool get _isAddable {
    final p = _activeSub.path;
    if (_isSpecialView) return false;
    return p != '/log' &&
        p != '/system/resource' &&
        p != '/system/routerboard' &&
        p != '/system/health' &&
        p != '/user/active';
  }

  Future<void> _loadActiveWinboxSubMenu({bool silent = false}) async {
    final sub = _activeSub;
    if (_isSpecialView) {
      if (!silent && mounted) setState(() => _isMenuLoading = false);
      return;
    }

    if (!silent && mounted) {
      setState(() => _isMenuLoading = true);
    }

    try {
      final printPath = sub.path.endsWith('/print') ? sub.path : '${sub.path}/print';
      final res = await widget.service.fetchMenu(printPath);
      if (mounted) {
        setState(() => _currentMenuItems = res);
      }
    } catch (_) {
      if (!silent && mounted) setState(() => _currentMenuItems = []);
    } finally {
      if (!silent && mounted) setState(() => _isMenuLoading = false);
    }
  }

  List<Map<String, String>> get _filteredItems {
    if (_searchQuery.isEmpty) return _currentMenuItems;
    final q = _searchQuery.toLowerCase();
    return _currentMenuItems.where((item) {
      return item.values.any((v) => v.toLowerCase().contains(q));
    }).toList();
  }

  void _onAdd() {
    final sub = _activeSub;
    final ifaces = (widget.data['interfaces'] as List<Map<String, String>>?) ?? [];

    if (sub.path == '/ip/address') {
      WinboxDialogs.showAddIpDialog(
        context: context,
        service: widget.service,
        interfaces: ifaces,
        onSuccess: _loadActiveWinboxSubMenu,
        showSnack: widget.showSnack,
      );
    } else if (sub.path == '/ip/route') {
      WinboxDialogs.showAddRouteDialog(
        context: context,
        service: widget.service,
        onSuccess: _loadActiveWinboxSubMenu,
        showSnack: widget.showSnack,
      );
    } else if (sub.path == '/queue/simple') {
      WinboxDialogs.showAddQueueDialog(
        context: context,
        service: widget.service,
        onSuccess: _loadActiveWinboxSubMenu,
        showSnack: widget.showSnack,
      );
    } else if (sub.path == '/ppp/secret') {
      WinboxDialogs.showAddPppSecretDialog(
        context: context,
        service: widget.service,
        onSuccess: _loadActiveWinboxSubMenu,
        showSnack: widget.showSnack,
      );
    } else {
      final role = widget.data['role'] as UserRoleInfo?;
      WinboxDialogs.showGenericAddDialog(
        context: context,
        service: widget.service,
        basePath: sub.path,
        canManageUsers: role?.canManageUsers ?? true,
        onSuccess: _loadActiveWinboxSubMenu,
        showSnack: widget.showSnack,
      );
    }
  }

  Future<void> _onToggleItem(Map<String, String> item, bool enable) async {
    final id = item['.id'];
    if (id == null) return;
    final name = item[_activeSub.primaryField] ?? item['name'] ?? item['address'] ?? 'Item';
    final actionText = enable ? 'mengaktifkan (enable)' : 'menonaktifkan (disable)';

    final confirm = await WinboxDialogs.showConfirmActionDialog(
      context: context,
      title: 'Konfirmasi',
      message: 'Apakah Anda yakin ingin $actionText "$name" (${_activeSub.path})?',
      confirmLabel: enable ? 'Yakin (Enable)' : 'Yakin (Disable)',
      confirmColor: enable ? const Color(0xFF34C759) : const Color(0xFFFF9500),
    );
    if (!confirm) return;

    try {
      final cmd = enable ? '${_activeSub.path}/enable' : '${_activeSub.path}/disable';
      await widget.service.executePath(cmd, params: {'.id': id});
      widget.showSnack(enable ? 'Item "$name" diaktifkan (Enabled).' : 'Item "$name" dinonaktifkan (Disabled).');
      await _loadActiveWinboxSubMenu();
    } catch (e) {
      widget.showSnack('Gagal: $e', isError: true);
    }
  }

  void _onCommentItem(Map<String, String> item) {
    final id = item['.id'];
    if (id == null) return;
    final name = item[_activeSub.primaryField] ?? item['name'] ?? item['address'] ?? 'Item';
    final commentCtrl = TextEditingController(text: item['comment'] ?? '');

    showDialog(
      context: context,
      builder: (ctx) => AlertDialog(
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(14)),
        title: Text('Winbox Comment ($name)'),
        content: TextField(
          controller: commentCtrl,
          autofocus: true,
          style: const TextStyle(fontSize: 13.5, color: Color(0xFF1C1C1E)),
          decoration: const InputDecoration(hintText: 'Tulis komentar untuk item ini...'),
        ),
        actions: [
          TextButton(onPressed: () => Navigator.pop(ctx), child: const Text('Batal')),
          ElevatedButton(
            style: ElevatedButton.styleFrom(
              backgroundColor: const Color(0xFF007AFF),
              foregroundColor: Colors.white,
              elevation: 0,
            ),
            onPressed: () async {
              final newComment = commentCtrl.text.trim();
              final oldComment = (item['comment'] ?? '').trim();
              if (newComment == oldComment) {
                Navigator.pop(ctx);
                return;
              }

              final confirm = await WinboxDialogs.showConfirmActionDialog(
                context: context,
                title: 'Konfirmasi Ubah Komentar',
                message: 'Apakah Anda sudah yakin ingin menerapkan komentar ini pada "$name"?\n\n• Komentar: "${newComment.isEmpty ? "(dikosongkan)" : newComment}"',
                confirmLabel: 'Yakin & Simpan',
              );
              if (!confirm) return;

              Navigator.pop(ctx);
              try {
                await widget.service.setComment(_activeSub.path, id, newComment);
                widget.showSnack('Komentar untuk "$name" berhasil disimpan.');
                await _loadActiveWinboxSubMenu();
              } catch (e) {
                widget.showSnack('Gagal: $e', isError: true);
              }
            },
            child: const Text('Simpan'),
          ),
        ],
      ),
    );
  }

  Future<void> _onRemoveItem(Map<String, String> item) async {
    final id = item['.id'];
    if (id == null) return;
    final name = item[_activeSub.primaryField] ?? item['name'] ?? item['address'] ?? 'Item';

    final confirm = await WinboxDialogs.showConfirmActionDialog(
      context: context,
      title: 'Hapus Item (Winbox Remove)?',
      message: 'Apakah Anda yakin ingin menghapus "$name" (${_activeSub.path}) secara permanen dari router?',
      confirmLabel: 'Hapus Permanen',
      isDestructive: true,
    );

    if (confirm) {
      try {
        await widget.service.executePath('${_activeSub.path}/remove', params: {'.id': id});
        widget.showSnack('Item "$name" berhasil dihapus.');
        await _loadActiveWinboxSubMenu();
      } catch (e) {
        widget.showSnack('Gagal: $e', isError: true);
      }
    }
  }

  void _showItemActionSheet(Map<String, String> item) {
    final name = item[_activeSub.primaryField] ?? item['name'] ?? item['address'] ?? 'Detail Item';
    final disabled = item['disabled'] == 'true';
    final hasId = item['.id'] != null;

    showModalBottomSheet(
      context: context,
      backgroundColor: Colors.white,
      shape: const RoundedRectangleBorder(borderRadius: BorderRadius.vertical(top: Radius.circular(20))),
      builder: (ctx) => SafeArea(
        child: Padding(
          padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 12),
          child: Column(
            mainAxisSize: MainAxisSize.min,
            crossAxisAlignment: CrossAxisAlignment.stretch,
            children: [
              // Handle Bar
              Center(
                child: Container(
                  width: 36,
                  height: 4,
                  decoration: BoxDecoration(color: const Color(0xFFD1D1D6), borderRadius: BorderRadius.circular(2)),
                ),
              ),
              const SizedBox(height: 12),

              // Title Header
              Row(
                children: [
                  Expanded(
                    child: Column(
                      crossAxisAlignment: CrossAxisAlignment.start,
                      children: [
                        Text(
                          name,
                          style: const TextStyle(fontSize: 16, fontWeight: FontWeight.bold, color: Color(0xFF1C1C1E)),
                          maxLines: 1,
                          overflow: TextOverflow.ellipsis,
                        ),
                        Text(
                          _activeSub.path,
                          style: const TextStyle(fontSize: 11, color: Color(0xFF8E8E93), fontFamily: 'monospace'),
                        ),
                      ],
                    ),
                  ),
                  Container(
                    padding: const EdgeInsets.symmetric(horizontal: 8, vertical: 3),
                    decoration: BoxDecoration(
                      color: disabled ? const Color(0xFFFF3B30).withValues(alpha: 0.1) : const Color(0xFF34C759).withValues(alpha: 0.1),
                      borderRadius: BorderRadius.circular(6),
                    ),
                    child: Text(
                      disabled ? 'NONAKTIF' : 'AKTIF',
                      style: TextStyle(
                        fontSize: 10,
                        fontWeight: FontWeight.bold,
                        color: disabled ? const Color(0xFFFF3B30) : const Color(0xFF34C759),
                      ),
                    ),
                  ),
                ],
              ),
              const SizedBox(height: 16),
              const Divider(height: 1, color: Color(0xFFE5E5EA)),
              const SizedBox(height: 8),

              // Action 1: Edit & Inspector
              ListTile(
                dense: true,
                contentPadding: const EdgeInsets.symmetric(horizontal: 4),
                leading: Container(
                  width: 36,
                  height: 36,
                  decoration: BoxDecoration(color: const Color(0xFF007AFF).withValues(alpha: 0.1), borderRadius: BorderRadius.circular(8)),
                  child: const Icon(Icons.edit_note_rounded, color: Color(0xFF007AFF), size: 20),
                ),
                title: const Text('Detail & Edit Properti', style: TextStyle(fontWeight: FontWeight.w600, fontSize: 14)),
                subtitle: const Text('Lihat dan ubah seluruh konfigurasi RouterOS', style: TextStyle(fontSize: 11.5, color: Color(0xFF8E8E93))),
                onTap: () {
                  Navigator.pop(ctx);
                  WinboxItemInspector.show(
                    context: context,
                    item: item,
                    basePath: _activeSub.path,
                    service: widget.service,
                    onRefresh: _loadActiveWinboxSubMenu,
                    showSnack: widget.showSnack,
                  );
                },
              ),

              // Action 2: Toggle Enable / Disable
              if (hasId)
                ListTile(
                  dense: true,
                  contentPadding: const EdgeInsets.symmetric(horizontal: 4),
                  leading: Container(
                    width: 36,
                    height: 36,
                    decoration: BoxDecoration(
                      color: (disabled ? const Color(0xFF34C759) : const Color(0xFFFF9500)).withValues(alpha: 0.1),
                      borderRadius: BorderRadius.circular(8),
                    ),
                    child: Icon(
                      disabled ? Icons.check_circle_outline_rounded : Icons.block_rounded,
                      color: disabled ? const Color(0xFF34C759) : const Color(0xFFFF9500),
                      size: 20,
                    ),
                  ),
                  title: Text(disabled ? 'Aktifkan Item (Enable)' : 'Nonaktifkan Item (Disable)', style: const TextStyle(fontWeight: FontWeight.w600, fontSize: 14)),
                  subtitle: Text(disabled ? 'Nyalakan kembali item ini di RouterOS' : 'Matikan sementara item ini tanpa menghapusnya', style: const TextStyle(fontSize: 11.5, color: Color(0xFF8E8E93))),
                  onTap: () {
                    Navigator.pop(ctx);
                    _onToggleItem(item, disabled);
                  },
                ),

              // Action 3: Comment
              if (hasId)
                ListTile(
                  dense: true,
                  contentPadding: const EdgeInsets.symmetric(horizontal: 4),
                  leading: Container(
                    width: 36,
                    height: 36,
                    decoration: BoxDecoration(color: const Color(0xFF5856D6).withValues(alpha: 0.1), borderRadius: BorderRadius.circular(8)),
                    child: const Icon(Icons.chat_bubble_outline_rounded, color: Color(0xFF5856D6), size: 18),
                  ),
                  title: const Text('Beri / Ubah Komentar', style: TextStyle(fontWeight: FontWeight.w600, fontSize: 14)),
                  subtitle: const Text('Tambahkan catatan identifikasi (;;; comment)', style: TextStyle(fontSize: 11.5, color: Color(0xFF8E8E93))),
                  onTap: () {
                    Navigator.pop(ctx);
                    _onCommentItem(item);
                  },
                ),

              // Action 4: Delete
              if (hasId)
                ListTile(
                  dense: true,
                  contentPadding: const EdgeInsets.symmetric(horizontal: 4),
                  leading: Container(
                    width: 36,
                    height: 36,
                    decoration: BoxDecoration(color: const Color(0xFFFF3B30).withValues(alpha: 0.1), borderRadius: BorderRadius.circular(8)),
                    child: const Icon(Icons.delete_outline_rounded, color: Color(0xFFFF3B30), size: 20),
                  ),
                  title: const Text('Hapus Item', style: TextStyle(fontWeight: FontWeight.w600, fontSize: 14, color: Color(0xFFFF3B30))),
                  subtitle: const Text('Hapus permanen item ini dari konfigurasi router', style: TextStyle(fontSize: 11.5, color: Color(0xFF8E8E93))),
                  onTap: () {
                    Navigator.pop(ctx);
                    _onRemoveItem(item);
                  },
                ),
            ],
          ),
        ),
      ),
    );
  }

  void _openMenuPicker() {
    showModalBottomSheet(
      context: context,
      isScrollControlled: true,
      backgroundColor: Colors.transparent,
      builder: (ctx) => _ModernMenuPickerSheet(
        activePath: _activeSub.path,
        onSelect: (catIdx, subIdx) {
          Navigator.pop(ctx);
          setState(() {
            _activeCategory = catIdx;
            _activeSubMenu = subIdx;
            _searchQuery = '';
            _isSearchOpen = false;
            _searchController.clear();
          });
          _loadActiveWinboxSubMenu();
        },
      ),
    );
  }

          decoration: const BoxDecoration(

                    )
                  : Row(

                                children: [
                                  Icon(_activeCat.icon, size: 18, color: const Color(0xFF007AFF)),
                                  const SizedBox(width: 8),

                                    child: Text(

                                      style: const TextStyle(
                                        fontSize: 16,
                                        fontWeight: FontWeight.bold,

                                  const SizedBox(width: 4),

  }

  Widget _buildBody() {
    final sub = _activeSub;

    if (sub.path == 'PING_TOOL') {
      return PingToolWidget(service: widget.service);
    } else if (sub.path == 'TRACEROUTE_TOOL') {
      return TracerouteToolWidget(service: widget.service);
    } else if (sub.path == 'PROFILE_TOOL') {
      return CpuProfileToolWidget(service: widget.service);
    } else if (sub.path == 'CLI_TOOL') {
      return CliTerminalWidget(service: widget.service);
    } else if (sub.path == 'SYSTEM_REBOOT' || sub.path == 'SYSTEM_SHUTDOWN') {
      return _buildSystemActionView(sub);
    }

    if (_isMenuLoading) {
      return const Center(child: CircularProgressIndicator(strokeWidth: 2));
    }

    final items = _filteredItems;

    return RefreshIndicator(
      onRefresh: _loadActiveWinboxSubMenu,
      child: items.isEmpty
          ? Center(
              child: SingleChildScrollView(
                physics: const AlwaysScrollableScrollPhysics(),
                child: Padding(
                  padding: const EdgeInsets.all(32),
                  child: Column(
                    mainAxisSize: MainAxisSize.min,
                    children: [
                      const Icon(Icons.inbox_rounded, size: 40, color: Color(0xFFC7C7CC)),
                      const SizedBox(height: 10),
                      Text(
                        _currentMenuItems.isEmpty ? 'Belum ada data di ${sub.title}.' : 'Tidak ada hasil untuk "$_searchQuery".',

class WinboxDialogs {
  static Future<bool> showConfirmActionDialog({
    required BuildContext context,
    required String title,
    required String message,
    String confirmLabel = 'Yakin',
    Color confirmColor = const Color(0xFF007AFF),
    bool isDestructive = false,
  }) async {
    final res = await showDialog<bool>(
      context: context,
      barrierDismissible: false,
      builder: (ctx) => AlertDialog(
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(16)),
        title: Text(
          title,
          style: const TextStyle(fontWeight: FontWeight.bold, fontSize: 16.5, color: Color(0xFF1C1C1E)),
        ),
        content: Text(
          message,
          style: const TextStyle(fontSize: 13.5, color: Color(0xFF3C3C43), height: 1.4),
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(ctx, false),
            child: const Text('Batal', style: TextStyle(color: Color(0xFF8E8E93), fontWeight: FontWeight.w500)),
          ),
          ElevatedButton(
            style: ElevatedButton.styleFrom(
              backgroundColor: isDestructive ? const Color(0xFFFF3B30) : confirmColor,
              foregroundColor: Colors.white,
              elevation: 0,
              shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(8)),
              padding: const EdgeInsets.symmetric(horizontal: 14, vertical: 8),
            ),
            onPressed: () => Navigator.pop(ctx, true),
            child: Text(confirmLabel, style: const TextStyle(fontWeight: FontWeight.w600, fontSize: 13)),
          ),
        ],
      ),
    );
    return res ?? false;
  }

  static Widget buildDialogField({
    required TextEditingController controller,
    required String label,
    required String placeholder,
    bool isMonospace = false,
  }) {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Text(
          label,
          style: const TextStyle(fontSize: 12, fontWeight: FontWeight.w500, color: Color(0xFF6C6C70)),
        ),
        const SizedBox(height: 5),
        Container(
          decoration: BoxDecoration(
            color: const Color(0xFFF2F2F7),
            borderRadius: BorderRadius.circular(8),
          ),
          child: TextField(
            controller: controller,
            style: TextStyle(
              fontSize: 13.5,
              fontFamily: isMonospace ? 'monospace' : null,
              color: const Color(0xFF1C1C1E),
            ),
            decoration: InputDecoration(
              isDense: true,
              hintText: placeholder,
              hintStyle: const TextStyle(fontSize: 13, color: Color(0xFFC7C7CC)),
              border: InputBorder.none,
              contentPadding: const EdgeInsets.symmetric(horizontal: 12, vertical: 9),
            ),
          ),
        ),
      ],
    );
  }

  static void showAddIpDialog({
    required BuildContext context,
    required MikroTikService service,
    required List<Map<String, String>> interfaces,
    required VoidCallback onSuccess,
    required Function(String, {bool isError}) showSnack,
  }) {
    final ipCtrl = TextEditingController();
    String selectedIface = interfaces.isNotEmpty ? (interfaces.first['name'] ?? 'ether1') : 'ether1';

    showDialog(
      context: context,
      builder: (ctx) => StatefulBuilder(
        builder: (ctx, setDlgState) => AlertDialog(
          shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(14)),
          title: const Text('Tambah IP Address'),
          content: Column(
            mainAxisSize: MainAxisSize.min,
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              buildDialogField(controller: ipCtrl, label: 'IP Address / Subnet (CIDR)', placeholder: '192.168.10.1/24'),
              const SizedBox(height: 12),
              const Text('Interface', style: TextStyle(fontSize: 12, color: Color(0xFF6C6C70))),
              const SizedBox(height: 6),
              Container(
                padding: const EdgeInsets.symmetric(horizontal: 10),
                decoration: BoxDecoration(color: const Color(0xFFF2F2F7), borderRadius: BorderRadius.circular(8)),
                child: DropdownButton<String>(
                  isExpanded: true,
                  underline: const SizedBox(),
                  value: selectedIface,
                  items: interfaces.map((i) {
                    final name = i['name'] ?? 'ether1';
                    return DropdownMenuItem(value: name, child: Text(name));
                  }).toList(),
                  onChanged: (v) {
                    if (v != null) setDlgState(() => selectedIface = v);
                  },
                ),
              ),
            ],
          ),
          actions: [
            TextButton(onPressed: () => Navigator.pop(ctx), child: const Text('Batal')),
            TextButton(
              onPressed: () async {
                final addr = ipCtrl.text.trim();
                if (addr.isEmpty) return;

                final confirm = await showConfirmActionDialog(
                  context: context,
                  title: 'Konfirmasi Tambah IP Address',
                  message: 'Apakah Anda sudah yakin ingin menambahkan IP Address "$addr" pada interface "$selectedIface"?',
                  confirmLabel: 'Yakin & Tambah',
                );
                if (!confirm) return;

                Navigator.pop(ctx);
                try {
                  await service.addIpAddress(address: addr, interface: selectedIface);
                  showSnack('IP $addr berhasil ditambahkan.');
                  onSuccess();
                } catch (e) {
                  showSnack('Gagal: $e', isError: true);
                }
              },
              child: const Text('Tambah', style: TextStyle(fontWeight: FontWeight.bold)),
            ),
          ],
        ),
      ),
    );
  }

  static void showAddQueueDialog({
    required BuildContext context,
    required MikroTikService service,
    required VoidCallback onSuccess,
    required Function(String, {bool isError}) showSnack,
  }) {
    final nameCtrl = TextEditingController(text: 'Limit-User');
    final targetCtrl = TextEditingController(text: '192.168.88.');
    final limitCtrl = TextEditingController(text: '5M/10M');

      context: context,
      builder: (ctx) => AlertDialog(
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(14)),
        title: const Text('Tambah Simple Queue'),

          mainAxisSize: MainAxisSize.min,
          children: [
            buildDialogField(controller: nameCtrl, label: 'Nama Rule', placeholder: 'User-1'),
            const SizedBox(height: 10),
            buildDialogField(controller: targetCtrl, label: 'Target IP / Subnet', placeholder: '192.168.88.50/32'),

          ],
        ),
        actions: [
          TextButton(onPressed: () => Navigator.pop(ctx), child: const Text('Batal')),
          TextButton(
            onPressed: () async {
              if (nameCtrl.text.isEmpty || targetCtrl.text.isEmpty) return;

              final confirm = await showConfirmActionDialog(
                context: context,
                title: 'Konfirmasi Tambah Simple Queue',
                message: 'Apakah Anda sudah yakin ingin menambahkan Simple Queue "${nameCtrl.text.trim()}" untuk target "${targetCtrl.text.trim()}" (Limit: ${limitCtrl.text.trim()})?',

              try {
                await service.addSimpleQueue(
                  name: nameCtrl.text.trim(),
                  target: targetCtrl.text.trim(),
                  maxLimit: limitCtrl.text.trim(),
                );

import 'package:flutter/material.dart';
import '../services/mikrotik_service.dart';
import 'winbox_dialogs.dart';

class WinboxItemInspector {
  static void show({
    required BuildContext context,
    required Map<String, String> item,
    required String basePath,
    required MikroTikService service,
    required VoidCallback onRefresh,
    required Function(String, {bool isError}) showSnack,
  }) {
    final id = item['.id'];
    final name = item['name'] ?? item['address'] ?? item['dst-address'] ?? item['chain'] ?? 'Detail Item';
    final disabled = item['disabled'] == 'true';

    // Editable controllers map
    final controllers = <String, TextEditingController>{};
    for (final entry in item.entries) {
      if (!entry.key.startsWith('.')) {
        controllers[entry.key] = TextEditingController(text: entry.value);
      }
    }

    showModalBottomSheet(
      context: context,
      isScrollControlled: true,
      backgroundColor: Colors.white,
      shape: const RoundedRectangleBorder(
        borderRadius: BorderRadius.vertical(top: Radius.circular(16)),
      ),
      builder: (ctx) => StatefulBuilder(
        builder: (ctx, setSheetState) => DraggableScrollableSheet(
          initialChildSize: 0.75,
          maxChildSize: 0.95,
          minChildSize: 0.45,
          expand: false,
          builder: (context, scrollCtrl) => Padding(
            padding: const EdgeInsets.fromLTRB(16, 10, 16, 20),
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.stretch,
              children: [
                // Winbox Grab & Title
                Center(
                  child: Container(
                    width: 36,
                    height: 4,
                    decoration: BoxDecoration(color: const Color(0xFFD1D1D6), borderRadius: BorderRadius.circular(2)),
                  ),
                ),
                const SizedBox(height: 10),
                Row(
                  children: [
                    Container(
                      padding: const EdgeInsets.symmetric(horizontal: 6, vertical: 2),
                      decoration: BoxDecoration(
                        color: const Color(0xFF007AFF).withValues(alpha: 0.12),
                        borderRadius: BorderRadius.circular(4),
                      ),
                      child: Text(
                        basePath,
                        style: const TextStyle(fontSize: 10, fontWeight: FontWeight.bold, color: Color(0xFF007AFF), fontFamily: 'monospace'),
                      ),
                    ),
                    const SizedBox(width: 8),
                    Expanded(
                      child: Text(
                        name,
                        style: const TextStyle(fontSize: 16, fontWeight: FontWeight.bold, color: Color(0xFF1C1C1E)),
                        overflow: TextOverflow.ellipsis,
                      ),
                    ),
                  ],
                ),
                const SizedBox(height: 10),

                // WINBOX ACTION BUTTONS BAR (OK, Cancel, Apply, Remove, Enable/Disable)
                Row(
                  children: [
                    // Apply Button
                    if (id != null)
                      Expanded(
                        child: TextButton(
                          style: TextButton.styleFrom(
                            backgroundColor: const Color(0xFF007AFF),
                            foregroundColor: Colors.white,
                            padding: const EdgeInsets.symmetric(vertical: 8),
                            shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(6)),
                          ),
                          onPressed: () async {
                            final updatedProps = <String, String>{};
                            controllers.forEach((k, v) {
                              final currentVal = v.text.trim();
                              final originalVal = (item[k] ?? '').trim();
                              if (currentVal != originalVal) {
                                updatedProps[k] = currentVal;
                              }
                            });

                            if (updatedProps.isEmpty) {
                              showSnack('Tidak ada perubahan konfigurasi yang dilakukan.');
                              return;
                            }

                            final changesSummary = updatedProps.entries.map((e) {
                              final oldV = item[e.key] ?? '(kosong)';
                              return '• ${e.key}: "$oldV" ➔ "${e.value}"';
                            }).join('\n');

                            final confirm = await WinboxDialogs.showConfirmActionDialog(
                              context: context,
                              title: 'Konfirmasi Perubahan Konfigurasi',
                              message: 'Apakah Anda sudah yakin ingin menerapkan perubahan konfigurasi berikut ke MikroTik?\n\n$changesSummary',
                              confirmLabel: 'Yakin & Terapkan',
                              confirmColor: const Color(0xFF007AFF),
                            );

                            if (!confirm) return;

                            // Tutup sheet terlebih dahulu agar notifikasi hijau langsung terlihat di layar
                            Navigator.pop(ctx);

                            try {
                              await service.setItemProperties(basePath, id, updatedProps);
                              showSnack('Perubahan konfigurasi untuk "$name" berhasil diterapkan!');
                              onRefresh();
                            } catch (e) {
                              showSnack('Gagal: $e', isError: true);
                            }
                          },
                          child: const Text('Apply', style: TextStyle(fontWeight: FontWeight.bold, fontSize: 12)),
                        ),
                      ),
                    const SizedBox(width: 6),
                    // Enable/Disable
                    if (id != null && item.containsKey('disabled'))
                      Expanded(
                        child: TextButton(
                          style: TextButton.styleFrom(
                            backgroundColor: disabled ? const Color(0xFFE8F5E9) : const Color(0xFFFFF3E0),
                            foregroundColor: disabled ? const Color(0xFF2E7D32) : const Color(0xFFE65100),
                            padding: const EdgeInsets.symmetric(vertical: 8),
                            shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(6)),
                          ),
                          onPressed: () async {
                            final willEnable = disabled;
                            final actionText = willEnable ? 'mengaktifkan (enable)' : 'menonaktifkan (disable)';

                            final confirm = await WinboxDialogs.showConfirmActionDialog(
                              context: context,
                              title: 'Konfirmasi',
                              message: 'Apakah Anda yakin ingin $actionText "$name" ($basePath)?',
                              confirmLabel: willEnable ? 'Yakin (Enable)' : 'Yakin (Disable)',
                              confirmColor: willEnable ? const Color(0xFF34C759) : const Color(0xFFFF9500),
                            );

                            if (!confirm) return;

                            Navigator.pop(ctx);
                            try {
                              final cmd = willEnable ? '$basePath/enable' : '$basePath/disable';
                              await service.executePath(cmd, params: {'.id': id});
                              showSnack(willEnable ? 'Item "$name" diaktifkan.' : 'Item "$name" dinonaktifkan.');
                              onRefresh();
                            } catch (e) {
                              showSnack('Gagal: $e', isError: true);
                            }
                          },
                          child: Text(disabled ? 'Enable' : 'Disable', style: const TextStyle(fontWeight: FontWeight.bold, fontSize: 12)),
                        ),
                      ),
                    const SizedBox(width: 6),
                    // Remove Button
                    if (id != null)
                      TextButton(
                        style: TextButton.styleFrom(
                          backgroundColor: const Color(0xFFFFEBEE),
                          foregroundColor: const Color(0xFFD32F2F),
                          padding: const EdgeInsets.symmetric(horizontal: 10, vertical: 8),
                          shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(6)),
                        ),

                          final confirm = await WinboxDialogs.showConfirmActionDialog(
                            context: context,
                            title: 'Hapus Item (Winbox Remove)?',
                            message: 'Apakah Anda yakin ingin menghapus "$name" ($basePath) secara permanen dari router?',
                            confirmLabel: 'Hapus Permanen',
                            isDestructive: true,
                          );

                          if (!confirm) return;

                          Navigator.pop(ctx);
                          try {
                            await service.executePath('$basePath/remove', params: {'.id': id});
                            showSnack('Item "$name" berhasil dihapus.');
                            onRefresh();
                          } catch (e) {
                            showSnack('Gagal: $e', isError: true);
                          }
                        },
                        child: const Text('Remove', style: TextStyle(fontWeight: FontWeight.bold, fontSize: 12)),
                      ),
                  ],
                ),
                const Divider(height: 16, color: Color(0xFFE5E5EA)),

                  child: ListView(
                    controller: scrollCtrl,
                    children: item.entries.map((e) {
                      final isInternal = e.key.startsWith('.');

                      final valLower = (ctrl?.text ?? e.value).trim().toLowerCase();

                            ),
                            Expanded(
                              child: isInternal || ctrl == null
                                  ? SelectableText(
                                      e.value,
                                      style: const TextStyle(fontSize: 13, color: Color(0xFF8E8E93), fontFamily: 'monospace'),
                                    )
                                  : isBoolean
                                      ? Row(
                                          children: [

                                                  }
                                                });

                                            ),
                                            const SizedBox(width: 8),
                                            Container(
                                              padding: const EdgeInsets.symmetric(horizontal: 8, vertical: 3),
                                              decoration: BoxDecoration(
                                                color: (valLower == 'true' || valLower == 'yes')
                                                    ? const Color(0xFF34C759).withValues(alpha: 0.12)
                                                    : const Color(0xFFFF3B30).withValues(alpha: 0.12),
                                                borderRadius: BorderRadius.circular(6),
                                              ),
                                              child: Text(
                                                ctrl.text,
                                                style: TextStyle(
                                                  fontSize: 12,
                                                  fontWeight: FontWeight.bold,
                                                  fontFamily: 'monospace',
                                                  color: (valLower == 'true' || valLower == 'yes')
                                                      ? const Color(0xFF34C759)
                                                      : const Color(0xFFFF3B30),
                                                ),
                                              ),
                                            ),
                                          ],
                                        )
                                      : Container(
                                          decoration: BoxDecoration(
                                            color: const Color(0xFFF2F2F7),
                                            borderRadius: BorderRadius.circular(6),
                                          ),
                                          child: TextField(
                                            controller: ctrl,
                                            style: TextStyle(
                                              fontSize: 13,
                                              color: const Color(0xFF1C1C1E),

              ],

import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import '../services/mikrotik_service.dart';

class PingToolWidget extends StatefulWidget {
  final MikroTikService service;

  const PingToolWidget({super.key, required this.service});

  @override
  State<PingToolWidget> createState() => _PingToolWidgetState();
}

class _PingToolWidgetState extends State<PingToolWidget> {
  final _targetController = TextEditingController(text: '8.8.8.8');
  final _countController = TextEditingController(text: '4');
  bool _isLoading = false;
  String _output = '';

  Future<void> _executePing() async {
    final target = _targetController.text.trim();
    final count = int.tryParse(_countController.text.trim()) ?? 4;
    if (target.isEmpty) return;

    setState(() {
      _isLoading = true;
      _output = 'Mengirim ICMP ping ke $target ($count paket)...';
    });

    try {
      final res = await widget.service.pingHost(target, count: count);
      setState(() {
        _output = res.map((r) {
          final host = r['host'] ?? target;
          final time = r['time'] ?? '-';
          final ttl = r['ttl'] ?? '-';
          final status = r['status'] ?? 'OK';
          final size = r['size'] ?? '56';
          return 'Reply from $host: bytes=$size time=$time ttl=$ttl status=$status';
        }).join('\n');
      });
    } catch (e) {
      setState(() => _output = 'Gagal ping: ${e.toString().replaceFirst("Exception: ", "")}');
    } finally {
      if (mounted) setState(() => _isLoading = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    return Padding(
      padding: const EdgeInsets.all(16),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.stretch,
        children: [
          Container(
            padding: const EdgeInsets.all(12),
            decoration: BoxDecoration(
              color: Colors.white,
              borderRadius: BorderRadius.circular(10),
              border: Border.all(color: const Color(0xFFE5E5EA), width: 0.5),
            ),
            child: Row(
              children: [
                Expanded(
                  flex: 3,
                  child: Container(
                    decoration: BoxDecoration(
                      color: const Color(0xFFF2F2F7),
                      borderRadius: BorderRadius.circular(8),
                    ),
                    child: TextField(
                      controller: _targetController,
                      style: const TextStyle(fontSize: 13, fontFamily: 'monospace', color: Color(0xFF1C1C1E)),
                      decoration: const InputDecoration(
                        filled: true,
                        fillColor: Color(0xFFF2F2F7),
                        hintText: 'Target IP / Hostname (misal: 8.8.8.8)',
                        isDense: true,
                        border: InputBorder.none,
                        contentPadding: EdgeInsets.symmetric(horizontal: 10, vertical: 8),
                      ),
                    ),
                  ),
                ),
                const SizedBox(width: 8),
                Expanded(
                  flex: 1,
                  child: Container(
                    decoration: BoxDecoration(
                      color: const Color(0xFFF2F2F7),
                      borderRadius: BorderRadius.circular(8),
                    ),
                    child: TextField(
                      controller: _countController,
                      keyboardType: TextInputType.number,
                      style: const TextStyle(fontSize: 13, color: Color(0xFF1C1C1E)),
                      decoration: const InputDecoration(
                        filled: true,
                        fillColor: Color(0xFFF2F2F7),
                        hintText: 'Count',
                        isDense: true,
                        border: InputBorder.none,
                        contentPadding: EdgeInsets.symmetric(horizontal: 10, vertical: 8),

class CliTerminalWidget extends StatefulWidget {
  final MikroTikService service;

  const CliTerminalWidget({super.key, required this.service});

  @override
  State<CliTerminalWidget> createState() => _CliTerminalWidgetState();
}

class _CliTerminalWidgetState extends State<CliTerminalWidget> {
  final _cliPathController = TextEditingController(text: '/ip/route/print');
  bool _isLoading = false;
  String _output = '';

  final List<String> _quickCommands = [
    '/system/resource/print',
    '/ip/address/print',
    '/interface/print',
    '/ip/route/print',
    '/ip/dhcp-server/lease/print',
    '/log/print',
    '/ip/firewall/filter/print',
  ];

  @override
  void dispose() {
    _cliPathController.dispose();
    super.dispose();
  }

  Future<void> _runCommand([String? overrideCmd]) async {
    final cmd = (overrideCmd ?? _cliPathController.text).trim();
    if (cmd.isEmpty) return;

    if (overrideCmd != null) {
      _cliPathController.text = overrideCmd;
    }

    setState(() => _isLoading = true);
    try {
      String path = cmd;
      Map<String, String>? params;

      if (cmd.contains(' ')) {
        final parts = cmd.split(RegExp(r'\s+'));
        if (parts.length > 1 && !parts[1].contains('=')) {
          if (parts.every((p) => !p.contains('='))) {
            path = parts.map((p) => p.replaceAll('/', '')).where((p) => p.isNotEmpty).join('/');
            path = '/$path';
          } else {
            path = parts[0];
            params = {};
            for (int i = 1; i < parts.length; i++) {
              final kv = parts[i].split('=');
              if (kv.length == 2) {
                params[kv[0]] = kv[1];
              } else if (kv.length == 1) {
                params['address'] = kv[0];
              }
            }
          }
        } else {
          path = parts[0];
          params = {};
          for (int i = 1; i < parts.length; i++) {
            final kv = parts[i].split('=');
            if (kv.length == 2) {
              params[kv[0]] = kv[1];
            }
          }
        }
      }

      final res = await widget.service.runCommand(path, params: params);
      setState(() {
        _output = res.isEmpty
            ? '(Perintah "$path" berhasil dieksekusi tanpa baris kembalian)'
            : res.map((row) => row.entries.map((e) => '${e.key}=${e.value}').join('  ')).join('\n\n');
      });
    } catch (e) {
      setState(() => _output = 'ERROR: ${e.toString().replaceFirst("Exception: ", "")}');
    } finally {
      if (mounted) setState(() => _isLoading = false);
    }
  }

  void _copyOutput() {
    if (_output.isEmpty) return;
    Clipboard.setData(ClipboardData(text: _output));
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: const Text('Output terminal berhasil disalin.'),
        behavior: SnackBarBehavior.floating,
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(10)),
        backgroundColor: const Color(0xFF34C759),
        duration: const Duration(seconds: 2),
      ),
    );
  }

  void _clearOutput() {
    setState(() => _output = '');
  }

  @override
  Widget build(BuildContext context) {
    return Padding(
      padding: const EdgeInsets.all(16),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.stretch,
        children: [
          // CLI COMMAND BAR (AUTHENTIC DARK TERMINAL THEME)
          Container(
            padding: const EdgeInsets.all(12),
            decoration: BoxDecoration(
              color: const Color(0xFF1C1C1E),
              borderRadius: BorderRadius.circular(12),
              border: Border.all(color: const Color(0xFF2C2C2E), width: 1),
            ),
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.stretch,
              children: [
                Row(
                  children: [
                    Expanded(
                      child: Container(
                        height: 42,
                        decoration: BoxDecoration(
                          color: const Color(0xFF2C2C2E),
                          borderRadius: BorderRadius.circular(8),
                          border: Border.all(color: const Color(0xFF3A3A3C), width: 0.8),
                        ),
                        child: TextField(
                          controller: _cliPathController,
                          cursorColor: const Color(0xFF34C759),
                          style: const TextStyle(
                            fontFamily: 'monospace',
                            color: Colors.white,
                            fontSize: 13,
                            fontWeight: FontWeight.w500,
                          ),
                          decoration: InputDecoration(
                            filled: true,
                            fillColor: const Color(0xFF2C2C2E),
                            hintText: 'Ketik perintah RouterOS...',
                            hintStyle: const TextStyle(color: Color(0xFF8E8E93), fontSize: 12, fontFamily: 'monospace'),
                            isDense: true,
                            prefixIcon: const Icon(Icons.terminal_rounded, size: 16, color: Color(0xFF34C759)),
                            prefixIconConstraints: const BoxConstraints(minWidth: 36, maxWidth: 36),

                                    onPressed: () {
                                      _cliPathController.clear();
                                      setState(() {});
                                    },

                                : null,

                            border: InputBorder.none,
