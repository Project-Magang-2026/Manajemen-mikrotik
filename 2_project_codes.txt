import 'package:flutter/material.dart';
import '../models/router_models.dart';
import '../services/mikrotik_service.dart';
import '../services/profile_storage_service.dart';
import '../services/router_discovery_service.dart';
import '../services/mikrotik_auto_discovery.dart';
import 'router_dashboard_screen.dart';

class ProfileListScreen extends StatefulWidget {
  const ProfileListScreen({super.key});

  @override
  State<ProfileListScreen> createState() => _ProfileListScreenState();
}

class _ProfileListScreenState extends State<ProfileListScreen> {
  final _storage = ProfileStorageService();
  final _discoveryService = RouterDiscoveryService();
  final _autoDiscovery = MikrotikAutoDiscovery();

  final _labelController = TextEditingController();
  final _ipController = TextEditingController();
  final _portController = TextEditingController(text: '8728');
  final _userController = TextEditingController(text: 'admin');
  final _passController = TextEditingController();

  List<RouterProfile> _savedProfiles = [];
  bool _isConnecting = false;
  bool _isScanning = false;
  bool _obscurePass = true;
  LoginRolePreference _selectedRolePreference = LoginRolePreference.auto;
  MikrotikDiscoveryResult? _detectedRouter;

  @override
  void initState() {
    super.initState();
    _loadSavedProfiles();
    _startZeroTouchDiscovery();
  }

  @override
  void dispose() {
    _labelController.dispose();
    _ipController.dispose();
    _portController.dispose();
    _userController.dispose();
    _passController.dispose();
    super.dispose();
  }

  Future<void> _loadSavedProfiles() async {
    final list = await _storage.getProfiles();
    if (mounted) setState(() => _savedProfiles = list);
  }

  void _startZeroTouchDiscovery() {
    _autoDiscovery.discoverGatewayRouter().then((result) {
      if (!mounted) return;
      if (result != null) {
        setState(() => _detectedRouter = result);
      }
    });
  }

  void _applyAutoDiscovery(MikrotikDiscoveryResult detected) {
    setState(() {
      _ipController.text = detected.gatewayIp;
      _portController.text = detected.apiPort.toString();
      if (_labelController.text.trim().isEmpty) {
        _labelController.text = 'MikroTik (${detected.gatewayIp})';
      }
    });
    _showSnack('Auto-fill: ${detected.gatewayIp}:${detected.apiPort}');
  }

  Future<void> _scanNetwork() async {
    setState(() => _isScanning = true);
    try {
      final found = await _discoveryService.discoverActiveRouter();
      if (!mounted) return;
      if (found != null) {
        setState(() {
          _ipController.text = found.ip;
          _portController.text = found.port.toString();
          if (_labelController.text.trim().isEmpty) {
            _labelController.text = found.label;
          }
        });
        _showSnack('Router ditemukan: ${found.ip}');
      } else {
        _showSnack('Tidak ada router terdeteksi.', isError: true);
      }
    } catch (e) {
      _showSnack('Gagal scan.', isError: true);
    } finally {
      if (mounted) setState(() => _isScanning = false);
    }
  }

  Future<void> _connectAndSave(RouterProfile profile) async {
    setState(() => _isConnecting = true);
    try {
      final service = MikroTikService(profile);
      final data = await service.fetchDashboardSummary();
      await _storage.saveProfile(profile, setAsActive: false);
      if (!mounted) return;
      Navigator.of(context).pushReplacement(
        MaterialPageRoute(
          builder: (_) => RouterDashboardScreen(
            profile: profile,
            initialData: data,
          ),
        ),
      );
    } catch (e) {
      _showSnack(
        'Gagal: ${e.toString().replaceFirst("Exception: ", "")}',
        isError: true,
      );
    } finally {
      if (mounted) setState(() => _isConnecting = false);
    }
  }

  void _submitNewConnection() {
    final host = _ipController.text.trim();
    final port = int.tryParse(_portController.text.trim()) ?? 8728;
    final user = _userController.text.trim();
    final pass = _passController.text;
    final label = _labelController.text.trim().isEmpty
        ? host
        : _labelController.text.trim();

    if (host.isEmpty || user.isEmpty) {
      _showSnack('IP dan Username wajib diisi.', isError: true);
      return;
    }

    final protocol = port == 443
        ? ApiProtocol.restApiHttps
        : port == 80
            ? ApiProtocol.restApiHttp
            : ApiProtocol.routerOsApi8728;

    final newProfile = RouterProfile(
      id: '${host}_${user}_$port',
      label: label,
      host: host,
      port: port,
      username: user,
      password: pass,
      protocol: protocol,
      rolePreference: _selectedRolePreference,
    );
    _connectAndSave(newProfile);
  }

  Future<void> _deleteProfile(RouterProfile profile) async {
    final confirm = await showDialog<bool>(
      context: context,
      builder: (ctx) => AlertDialog(
        title: const Text('Hapus Profil?'),
        content: Text('Hapus "${profile.label}"?'),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(ctx, false),
            child: const Text('Batal'),
          ),
          FilledButton(
            style: FilledButton.styleFrom(
              backgroundColor: const Color(0xFFFF3B30),
            ),
            onPressed: () => Navigator.pop(ctx, true),
            child: const Text('Hapus'),
          ),
        ],
      ),
    );
    if (confirm != true) return;
    await _storage.deleteProfile(profile.id);
    await _loadSavedProfiles();
    _showSnack('Profil dihapus.');
  }

  void _showSnack(String msg, {bool isError = false}) {
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(msg, style: const TextStyle(fontSize: 13)),
        backgroundColor:
            isError ? const Color(0xFFFF3B30) : const Color(0xFF34C759),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: SafeArea(
        child: SingleChildScrollView(
          padding: const EdgeInsets.all(20),
          child: ConstrainedBox(
            constraints: const BoxConstraints(maxWidth: 480),
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.stretch,
              children: [
                const SizedBox(height: 8),
                _buildHeader(),
                const SizedBox(height: 24),
                if (_detectedRouter != null) ...[
                  _buildDiscoveryBanner(_detectedRouter!),
                  const SizedBox(height: 20),
                ],
                if (_savedProfiles.isNotEmpty) ...[
                  _buildSectionLabel('ROUTER TERSIMPAN'),
                  const SizedBox(height: 8),
                  ..._savedProfiles.map((p) => _buildProfileCard(p)),
                  const SizedBox(height: 24),
                ],
                Row(
                  mainAxisAlignment: MainAxisAlignment.spaceBetween,
                  children: [
                    _buildSectionLabel('HUBUNGKAN ROUTER'),
                    TextButton.icon(
                      onPressed:
                          _isScanning || _isConnecting ? null : _scanNetwork,
                      icon: _isScanning
                          ? const SizedBox(
                              width: 14,
                              height: 14,
                              child:
                                  CircularProgressIndicator(strokeWidth: 2),
                            )
                          : const Icon(Icons.wifi_find_rounded, size: 16),
                      label: Text(
                          _isScanning ? 'Scanning...' : 'Scan'),
                      style: TextButton.styleFrom(
                        foregroundColor: const Color(0xFF0071E3),
                        textStyle: const TextStyle(
                            fontSize: 12, fontWeight: FontWeight.w600),
                        padding: const EdgeInsets.symmetric(
                            horizontal: 8, vertical: 4),
                        minimumSize: Size.zero,
                        tapTargetSize: MaterialTapTargetSize.shrinkWrap,
                      ),
                    ),
                  ],
                ),
                const SizedBox(height: 8),
                _buildConnectionForm(),
                const SizedBox(height: 24),
                const Row(
                  mainAxisAlignment: MainAxisAlignment.center,
                  children: [
                    Icon(Icons.lock_outline_rounded,
                        size: 13, color: Color(0xFFAEAEB2)),
                    SizedBox(width: 6),
                    Text(
                      'Sesi otomatis berakhir saat app ditutup',
                      style: TextStyle(
                          fontSize: 11, color: Color(0xFFAEAEB2)),
                    ),
                  ],
                ),
                const SizedBox(height: 16),
              ],
            ),
          ),
        ),
      ),
    );
  }

  Widget _buildHeader() {
    return Row(
      children: [
        Container(
          width: 48,
          height: 48,
          decoration: BoxDecoration(
            color: const Color(0xFF0071E3).withOpacity(0.1),
            borderRadius: BorderRadius.circular(12),
          ),
          child: const Icon(Icons.router_rounded,
              color: Color(0xFF0071E3), size: 26),
        ),
        const SizedBox(width: 14),
        const Expanded(
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              Text(
                'Pocket Winbox',
                style: TextStyle(
                  fontSize: 22,
                  fontWeight: FontWeight.w700,
                  color: Color(0xFF1D1D1F),
                  letterSpacing: -0.5,
                ),
              ),
              SizedBox(height: 2),
              Text(
                'MikroTik Network Manager',
                style:
                    TextStyle(fontSize: 13, color: Color(0xFF86868B)),
              ),
            ],
          ),
        ),
      ],
    );
  }

    return Container(
      padding: const EdgeInsets.all(14),
      decoration: BoxDecoration(
        color: const Color(0xFF0071E3).withOpacity(0.06),
        borderRadius: BorderRadius.circular(12),
        border: Border.all(
            color: const Color(0xFF0071E3).withOpacity(0.2)),
      ),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Row(
            children: [
              const Icon(Icons.auto_awesome_rounded,
                  size: 18, color: Color(0xFF0071E3)),
              const SizedBox(width: 8),
              const Expanded(
                child: Text(
                  'Router MikroTik Terdeteksi',
                  style: TextStyle(
                      fontSize: 13, fontWeight: FontWeight.w600),
                ),
              ),
              Text(

                style: const TextStyle(
                  fontSize: 12,

                ),
              ),
            ],
          ),
          const SizedBox(height: 10),
          SizedBox(

              label: const Text('Isi Otomatis',
                  style: TextStyle(fontSize: 12)),
              style: FilledButton.styleFrom(

                shape: RoundedRectangleBorder(

  }

  }

                              fontWeight: FontWeight.w600),

                      ],
                    ),

          ),
        ],
      ),

  }

  Widget _buildField({
    required TextEditingController controller,
    required String label,
    required String hint,
    IconData? icon,
    bool obscure = false,
    bool mono = false,
    Widget? suffix,
    TextInputType keyboard = TextInputType.text,
  }) {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Text(
          label,
          style: const TextStyle(
              fontSize: 12,
              fontWeight: FontWeight.w500,
              color: Color(0xFF86868B)),
        ),
        const SizedBox(height: 6),
        TextField(
          controller: controller,
          obscureText: obscure,
          keyboardType: keyboard,
          style: TextStyle(
            fontSize: 14,
            fontFamily: mono ? 'monospace' : null,
          ),
          decoration: InputDecoration(
            hintText: hint,
            prefixIcon: icon != null
                ? Icon(icon,
                    size: 18, color: const Color(0xFFAEAEB2))
                : null,
            suffixIcon: suffix,
            isDense: true,
          ),
        ),
      ],
    );
  }
}

import 'dart:async';
import 'package:flutter/material.dart';
import '../models/router_models.dart';
import '../services/mikrotik_service.dart';
import '../services/profile_storage_service.dart';
import '../widgets/customer_tab_view.dart';
import '../widgets/winbox_panel_view.dart';
import 'profile_list_screen.dart';

class RouterDashboardScreen extends StatefulWidget {
  final RouterProfile profile;
  final Map<String, dynamic> initialData;

  const RouterDashboardScreen({
    super.key,
    required this.profile,
    required this.initialData,
  });

  @override
  State<RouterDashboardScreen> createState() =>
      _RouterDashboardScreenState();
}

class _RouterDashboardScreenState extends State<RouterDashboardScreen> {
  late MikroTikService _service;
  late Map<String, dynamic> _data;
  bool _isLoading = false;
  int _selectedTab = 0;
  bool _isOnline = true;
  Timer? _heartbeatTimer;
  StreamSubscription<bool>? _connectionSub;

  @override
  void initState() {
    super.initState();
    _service = MikroTikService(widget.profile);
    _data = widget.initialData;
    _subscribeConnection();
    _startHeartbeat();
  }

  @override
  void dispose() {
    _connectionSub?.cancel();
    _heartbeatTimer?.cancel();
    _service.dispose();
    super.dispose();
  }

  void _subscribeConnection() {
    _connectionSub =
        _service.connectionStateStream.listen((connected) {
      if (!mounted) return;
      if (_isOnline != connected) {
        setState(() => _isOnline = connected);
      }
    });
  }

  void _startHeartbeat() {
    _heartbeatTimer =
        Timer.periodic(const Duration(milliseconds: 1500), (_) async {
      if (!mounted || _isLoading) return;
      final alive = await _service.checkConnection();
      if (!mounted) return;
      if (alive != _isOnline) {
        setState(() => _isOnline = alive);
      }
      if (alive) {
        try {
          final res =
              await _service.fetchMenu('/system/resource/print');
          if (mounted && res.isNotEmpty) {
            setState(() => _data['resource'] = res.first);
          }
        } catch (_) {}
      }
    });
  }

  UserRoleInfo get _role => _data['role'] as UserRoleInfo;

  Future<void> _refresh() async {
    setState(() => _isLoading = true);
    try {
      final fresh = await _service.fetchDashboardSummary();
      if (mounted) {
        setState(() {
          _data = fresh;
          _isOnline = true;
          if (_role.isUser) _selectedTab = 0;
        });
      }
    } catch (e) {
      if (mounted) setState(() => _isOnline = false);
      _showSnack('Gagal refresh.', isError: true);
    } finally {
      if (mounted) setState(() => _isLoading = false);
    }
  }

  Future<void> _switchRouter() async {
    _heartbeatTimer?.cancel();
    await _service.dispose();
    await ProfileStorageService().setActiveProfileId(null);
    if (!mounted) return;
    Navigator.of(context).pushReplacement(
      MaterialPageRoute(builder: (_) => const ProfileListScreen()),
    );
  }

  void _showSnack(String msg, {bool isError = false}) {
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(msg, style: const TextStyle(fontSize: 13)),
        backgroundColor: isError
            ? const Color(0xFFFF3B30)
            : const Color(0xFF34C759),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    final isTechnician = _role.isTechnician;
    final activeTab = isTechnician ? _selectedTab : 0;

    return Scaffold(
      appBar: AppBar(
        titleSpacing: 16,
        title: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          mainAxisSize: MainAxisSize.min,
          children: [
            Row(
              children: [
                Flexible(
                  child: Text(
                    _data['identity']?.toString() ??
                        widget.profile.label,
                    style: const TextStyle(
                        fontSize: 16,
                        fontWeight: FontWeight.w700),
                    maxLines: 1,
                    overflow: TextOverflow.ellipsis,
                  ),
                ),
                const SizedBox(width: 8),
                _buildStatusPill(),
              ],
            ),
            const SizedBox(height: 2),
            Text(
              _role.isUser
                  ? 'MODE USER'
                  : '${widget.profile.host} • ${_role.roleBadge}',
              style: const TextStyle(
                  fontSize: 11, color: Color(0xFF86868B)),
              maxLines: 1,
              overflow: TextOverflow.ellipsis,
            ),
          ],
        ),
        actions: [
          IconButton(
            icon: _isLoading
                ? const SizedBox(
                    width: 16,
                    height: 16,
                    child:
                        CircularProgressIndicator(strokeWidth: 2))
                : const Icon(Icons.refresh_rounded, size: 20),
            onPressed: _isLoading ? null : _refresh,
          ),
          IconButton(
            icon: const Icon(Icons.swap_horiz_rounded, size: 20),
            onPressed: _switchRouter,
          ),
          const SizedBox(width: 4),
        ],
      ),
      body: Column(
        children: [
          if (!_isOnline) _buildOfflineBanner(),
          if (isTechnician) ...[
            Padding(
              padding: const EdgeInsets.symmetric(
                  horizontal: 16, vertical: 8),
              child: _buildTabSwitcher(),
            ),
          ],
          Expanded(
            child: activeTab == 0
                ? CustomerTabView(
                    data: _data,
                    role: _role,
                    service: _service,
                    onRefresh: _refresh,
                    showSnack: _showSnack,
                  )
                : WinboxPanelView(
                    service: _service,
                    data: _data,
                    showSnack: _showSnack,
                  ),
          ),
        ],
      ),
    );
  }

  Widget _buildStatusPill() {
    return Container(
      padding:
          const EdgeInsets.symmetric(horizontal: 6, vertical: 2),

  Future<void> _runPingTest() async {
    setState(() {
      _isTestingPing = true;
      _pingStatus = 'Menguji koneksi...';
      _pingColor = const Color(0xFF0071E3);
      _pingIcon = Icons.sync_rounded;
    });
    try {
      final res = await widget.service.pingHost('8.8.8.8', count: 2);
      if (!mounted) return;
      int? latency;
      bool received = false;
      for (final r in res) {
        final timeStr = r['time'] ?? '';
        if (timeStr.isNotEmpty) {
          received = true;
          final match = RegExp(r'(\d+)ms').firstMatch(timeStr);
          if (match != null) latency = int.tryParse(match.group(1)!);
          break;
        }
      }
      setState(() {
        if (!received) {
          _pingStatus = 'Tidak ada internet';
          _pingColor = const Color(0xFFFF3B30);
          _pingIcon = Icons.cloud_off_rounded;
        } else if (latency != null && latency < 50) {
          _pingStatus = 'Sangat Lancar ($latency ms)';
          _pingColor = const Color(0xFF34C759);
          _pingIcon = Icons.check_circle_rounded;
        } else if (latency != null && latency < 150) {
          _pingStatus = 'Stabil ($latency ms)';
          _pingColor = const Color(0xFF0071E3);
          _pingIcon = Icons.wifi_rounded;
        } else {
          _pingStatus = 'Agak Lambat ($latency ms)';
          _pingColor = const Color(0xFFFF9500);
          _pingIcon = Icons.warning_amber_rounded;
        }
      });
    } catch (_) {
      if (mounted) {
        setState(() {
          _pingStatus = 'Gagal tes koneksi';
          _pingColor = const Color(0xFFFF3B30);
          _pingIcon = Icons.error_outline_rounded;
        });
      }
    } finally {
      if (mounted) setState(() => _isTestingPing = false);
    }
  }

  List<ConnectedDevice> get _allDevices {
    final devList = (widget.data['devices'] as List<ConnectedDevice>?) ?? [];
    if (devList.isNotEmpty) return devList;
    final staticList = (widget.data['staticDevices'] as List<StaticHostDevice>?) ?? [];
    return staticList
        .map((s) => ConnectedDevice(
              macAddress: s.macAddress,
              ipAddress: s.ipAddress,
              hostName: s.customerName,
              source: s.interfaceName,
              isBlocked: s.isBlocked,
            ))
        .toList();
  }

  List<ConnectedDevice> get _filteredDevices {
    var list = _allDevices;
    if (_deviceFilterMode == 1) list = list.where((d) => !d.isBlocked).toList();
    if (_deviceFilterMode == 2) list = list.where((d) => d.isBlocked).toList();
    if (_searchQuery.isNotEmpty) {
      final q = _searchQuery.toLowerCase();
      list = list.where((d) {
        return d.hostName.toLowerCase().contains(q) ||
            d.ipAddress.toLowerCase().contains(q) ||
            d.macAddress.toLowerCase().contains(q);
      }).toList();
    }
    return list;
  }

  Future<void> _toggleBlock(ConnectedDevice device) async {
    final willBlock = !device.isBlocked;
    final confirm = await showDialog<bool>(
      context: context,
      builder: (ctx) => AlertDialog(
        title: Text(willBlock ? 'Blokir Perangkat?' : 'Buka Blokir?'),
        content: Text(
          willBlock
              ? 'Blokir ${device.hostName} (${device.macAddress})?'
              : 'Buka blokir ${device.hostName}?',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(ctx, false),
            child: const Text('Batal'),
          ),
          FilledButton(
            style: FilledButton.styleFrom(
              backgroundColor: willBlock ? const Color(0xFFFF3B30) : const Color(0xFF34C759),
            ),
            onPressed: () => Navigator.pop(ctx, true),
            child: Text(willBlock ? 'Blokir' : 'Buka'),
          ),
        ],
      ),
    );
    if (confirm != true) return;
    try {
      await widget.service.toggleBlockMac(device.macAddress, block: willBlock, label: device.hostName);
      if (device.ipAddress.isNotEmpty && device.ipAddress != '-' && device.ipAddress != 'Terblokir') {
        await widget.service.toggleBlockIp(device.ipAddress, block: willBlock, label: device.hostName);
      }
      widget.showSnack(willBlock ? 'Diblokir.' : 'Blokir dibuka.');
      widget.onRefresh();
    } catch (e) {
      widget.showSnack('Gagal: ${e.toString()}', isError: true);
    }
  }

          Text('Belum ada perangkat terdeteksi', style: TextStyle(color: Color(0xFF86868B), fontSize: 13)),
        ],
      ),
    );
  }

  void _showEditWifiDialog(WifiInterfaceInfo wifi) {
    final ssidCtrl = TextEditingController(text: wifi.ssid);
    final passCtrl = TextEditingController(text: wifi.password);

    showDialog(
      context: context,
      builder: (ctx) => AlertDialog(
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(14)),
        title: Text('Atur Wi-Fi (${wifi.name})'),
        content: Column(
          mainAxisSize: MainAxisSize.min,
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            const Text('Nama Wi-Fi (SSID)', style: TextStyle(fontSize: 12, fontWeight: FontWeight.w500, color: Color(0xFF6C6C70))),
            const SizedBox(height: 6),
            TextField(
              controller: ssidCtrl,
              decoration: InputDecoration(
                filled: true,
                fillColor: const Color(0xFFF2F2F7),
                contentPadding: const EdgeInsets.symmetric(horizontal: 12, vertical: 10),
                border: OutlineInputBorder(borderRadius: BorderRadius.circular(8), borderSide: BorderSide.none),
              ),
            ),
            const SizedBox(height: 12),
            const Text('Password Baru', style: TextStyle(fontSize: 12, fontWeight: FontWeight.w500, color: Color(0xFF6C6C70))),
            const SizedBox(height: 6),
            TextField(
              controller: passCtrl,
              decoration: InputDecoration(
                hintText: 'Min. 8 karakter',
                filled: true,
                fillColor: const Color(0xFFF2F2F7),
                contentPadding: const EdgeInsets.symmetric(horizontal: 12, vertical: 10),
                border: OutlineInputBorder(borderRadius: BorderRadius.circular(8), borderSide: BorderSide.none),
              ),
            ),
          ],
        ),
        actions: [
          TextButton(onPressed: () => Navigator.pop(ctx), child: const Text('Batal')),
          TextButton(
            onPressed: () async {
              final newSsid = ssidCtrl.text.trim();
              final newPass = passCtrl.text;
              if (newSsid.isEmpty) return;
              Navigator.pop(ctx);
              try {
                await widget.service.updateWifiConfig(
                  wifi: wifi,
                  newSsid: newSsid,
                  newPassword: newPass,
                );
                widget.showSnack('Pengaturan Wi-Fi berhasil diperbarui!');
                widget.onRefresh();
              } catch (e) {
                widget.showSnack('Gagal: $e', isError: true);
              }
            },
            child: const Text('Simpan', style: TextStyle(fontWeight: FontWeight.bold, color: Color(0xFF0071E3))),
          ),
        ],
      ),
    );
  }
